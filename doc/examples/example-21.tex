\documentclass{beamer}
\usepackage[kernel = lua]{../../latex/pgf-go}

% maeda-segoe.sgf: Maeda Nobuaki against Segoe Kensaku, 1940. problem.sgf: a problem with its solution.
\pgfgodefinesgf{game}{maeda-segoe.sgf}
\pgfgodefinesgf{problem}{problem.sgf}
\pgfgoset{profile = shell, goban = {label = {at = {south, west}}}}


\begin{document}
	\begin{frame}{A lesson with pgf-go}
		\begin{itemize}
			\item The opening of a game, move by move.
			\item The whole game in diagrams.
			\item A problem, and then its solution.
		\end{itemize}
	\end{frame}

	\begin{frame}[t]{The opening}
		\begin{columns}[T]
			\column{0.55\textwidth}
				\foreach \k in {1, ..., 12} {
					\only<\k>{
						\begin{goban}[sgf size = game, scale = 0.55]
							\sgf[to = \k, last mark = circle]{game}
						\end{goban}
					}
				}
			\column{0.4\textwidth}
				\small
				Black: \sgfinfo{game}{PB} (\sgfinfo{game}{BR})\par
				White: \sgfinfo{game}{PW} (\sgfinfo{game}{WR})\par
				Komi: \sgfinfo{game}{KM}\par
				\medskip
				\foreach \k in {1, ..., 12} {\only<\k>{Move \k}}
		\end{columns}
	\end{frame}

	\begin{frame}{The game in diagrams}
		\sgfdiagrams[step = 100, to = 200, layout = {columns = 2, fit}]{game}
	\end{frame}

	\begin{frame}[t]{A problem}
		\only<1>{\sgfproblem[solutions = false, goban = {scale = 2}]{problem}}
		\only<2>{\sgfproblem[statement = false, layout = {columns = 4, fit}]{problem}}
	\end{frame}
\end{document}
