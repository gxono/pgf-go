\section{Definiciones de piedras y de tableros}


\subsection{Piedras y jugadores}
	\subsubsection{pgfgo}
	
	\begin{center}
		\pgfgoset{profile = pgfgo}
		\begin{goban}[10x2, label = {at = {all}}, scale = 2]
			\stone{A1}
			\moves[b]{A:D2}
			\stone[w, mark = {triangle}]{D1}
			\moves*[b]{G1--J1|-G2}
			\mark[b, t-circle]{E1}
			\mark[w, t-square]{E2}
		\end{goban}
	\end{center}
	
	
	Declaramos todos los jugadores y comenzamos definiendo la apariencia del neutral.
	
	\begin{codeexample}[code only]
		\pgfgodefineplayer{pgfgo-neutral}{
			stone = {
				line = {
					true,
					color = black,
					width = 0.3pt,
					opacity = 1,
				},
				fill = {
					true,
					color = gray,
					opacity = 1,
				},
				radius = 0.17cm,
				scale = 1,
				drop shadow = {true, opacity = 0.5, offset = 1pt, angle = -45},
				shine = {true, angle = 45, color = wainshade}
			},
			move = {
				label = {
					color = black,
					format = arabic,
				}
			},
			mark = {
				line = {
					color = black,
					width = 1pt,
				},
				fill = {true, color = gray},
				label = {
					color = black,
					format = arabic
				}
			}
		}
	\end{codeexample}
	
	
	Algunas cosas como el grosor de linea, opacidades y dem\'as, deseo que se conserven en todas las piedras, as\'i que copio la plantilla del jugador neutral.
	
	\begin{codeexample}[code only]
		\pgfgodefineplayer[from = pgfgo-neutral]{pgfgo-white}{
			stone = {
				fill = {color = wainwhite},
				shine = {color = white}
			},
			move = {
				label = {
					color = black
				}
			},
			mark = {
				line 	= {color = black},
				label = {color = black}
			}
		}

		\pgfgodefineplayer[from = pgfgo-neutral]{pgfgo-black}{
			stone = {
				fill = {
					color = wainblack
				}
			},
			move = {
				label = {
					color = pgfgo
				}
			},
			mark = {
				line = {
					color = pgfgo,
				},
				label = {color = black}
			}
		}
	\end{codeexample}
