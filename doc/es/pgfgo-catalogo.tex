\subsection{Perfiles}
\subsubsection{default}
\begin{codeexample}[width=4cm]
	\pgfgoset{profile = default}
	
	\begin{goban}[partial = A1:G3]
		\moves*[from = 1]{A:F 3}
		\stones[w]{A1:2}
		\stones[b]{B1:2}
		\stones[n]{C1:2}
		\stone[w, mark = circle]{D1}
		\mark[w, circle]{D2}
		\stone[b, mark = circle]{E1}
		\mark[b, circle]{E2}
		\stone[n, mark = circle]{F1}
		\mark[n, circle]{F2}
		\mark[w, t-square]{F1}
		\mark[b, t-circle]{F1}
		\mark[n, t-square]{F1}
	\end{goban}
\end{codeexample}
