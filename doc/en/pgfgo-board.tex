\section{The Board}
\label{sec:board}

When we refer to the "board," we mean an environment or \emph{scope} in which both the goban (the game grid) and its contents—stones, coordinate labels, and other relevant information—are rendered. This environment is designed to be flexible and customizable, allowing users to adapt its appearance and functionality according to the needs of the diagram, whether to represent full games, specific sections, or even add visual annotations.

\begin{environment}{{goban}\opt{\oarg{options}}}
	
	Initializes the goban environment. By default, it draws a square 9×9 intersection board with a visible border, white background, and no labels on the sides.
	
	\begin{codeexample}[width=5cm]
		\begin{goban}
		\end{goban}
	\end{codeexample}
	
	Since the board is the only macro that utilizes the TikZ layer, each intersection can be accessed using its alphanumeric coordinate (as in \pgfname) or numeric values in the \tikzname\ coordinate system. This allows combining Go’s intuitive syntax with TikZ’s graphical capabilities, which is especially useful for adding custom elements, such as arrows or marks, directly onto the board.
	
	\begin{codeexample}[width=5.85cm]
		\begin{goban}[19, partial = J1:S9,
			label = {at = {north, east}}]
			\stones[b] {K8,R8,Q5,R5,O4,Q4,L3,O3,M2,N2}
			\mark[black, circle] {K8}
			\stones[white] {K6,R4,R3,P3,N1,O2,Q2}
			
			\tikzset{flecha/.style = {-{latex[]}, line width = 2pt, blue}}
			\draw[flecha] (12,8)--(17,8);
		\end{goban}
	\end{codeexample}
	
	\label{subsec:usei}\begin{key}{/pgfgo/goban/use i=\meta{boolean} (default true, initially true)}
		
		Enables or disables the use of the letter "I" in the board’s coordinates and labels. When disabled ("false"), any reference to "I" is interpreted as "J".
		
		\begin{codeexample}[]
			\pgfgoset{goban = {13, label = {at = north}}}
			
			\begin{goban}[label = {at = west}, partial = H1:M3]
				\stone[b]{J2}
			\end{goban}
			
			\begin{goban}[use i = false, partial = H1:N3]
				\stone[b]{J2}
			\end{goban}
			
			\begin{goban}[use i = false, label = {at = east}, partial = H1:N3]
				\stone[white, mark = {custom = !}] {I2} % If it’s I, it shifts to J.
			\end{goban}
		\end{codeexample}
	\end{key}
	
	\begin{key}{/pgfgo/goban/size=\meta{nlines or nlines |x| nlines} (initially 9)}
		
		Defines the number of vertical and horizontal lines on the board. The word "size" can be omitted, and the value specified directly. If a single integer (e.g., |9|) is provided, the board will be square (9×9). If an expression like |5x3| is used, a rectangular board with 5 columns and 3 rows will be created.
		
		\begin{codeexample}[width=6.5cm]
			\begin{goban}[5]
			\end{goban}
			
			\begin{goban}[5x3]
			\end{goban}
		\end{codeexample}
	\end{key}
	
	\begin{key}{/pgfgo/goban/sgf view=\meta{file}}
		Like |sgf size| and, if the file has a view (|VW|), it only shows that part of the board, like |partial|. The view can change during the game; the one of move |sgf view move| is used.
	\end{key}

	\begin{key}{/pgfgo/goban/sgf view move=\meta{number} (initially 0)}
		The move whose view |sgf view| uses. A |VW| holds from the move of its node (or from the next one, if the node has no move) until the next |VW|, and |VW[]| shows the whole board again. With 0, it is the view at the start. It can go before or after |sgf view|. |\sgfdiagrams|, |\sgfanimation| and |\sgfproblem| choose the view by themselves.

		\begin{codeexample}[code only]
			\begin{goban}[sgf view = game.sgf, sgf view move = 120]
				\sgf[from = 101, to = 120]{game.sgf}
			\end{goban}
		\end{codeexample}
	\end{key}

	\begin{key}{/pgfgo/goban/sgf size=\meta{file}}
		Takes the size of the board from an SGF game (its |SZ| property). Only with the Lua kernel (p.~\pageref{subsec:kernel}).

		\begin{codeexample}[]
			\begin{goban}[sgf size = variations.sgf, label = {at = {north, west}}]
				\sgf{variations.sgf}
			\end{goban}
		\end{codeexample}
	\end{key}

	\begin{key}{/pgfgo/goban/scale=\meta{float} (initially 1)}
		
		Controls the global scale of the board, affecting both its overall dimensions and the thickness of certain elements (e.g., the "line width" of the lines).
		
		\begin{codeexample}[width=5.5cm]
			\begin{goban}[2, scale = 2]
				\stone[b, mark]{A1}
			\end{goban}
			
			\begin{goban}[2, scale = 1]
				\stone[b, mark]{A1}
			\end{goban}
			
			\begin{goban}[2, scale = 0.5]
				\stone[b, mark]{A1}
			\end{goban}
		\end{codeexample}
	\end{key}
	
	\subsection{Grid}
	
	The grid refers to the central lines that form the board’s intersections.
	
	\begin{key}{/pgfgo/goban/grid=\meta{options} (default true)}
		Controls the behavior and appearance of the board’s central lines. This set of options allows modifying their visibility, color, thickness, and spacing, offering a high degree of customization.
	\end{key}
	
	\begin{stylekey}{/pgfgo/goban/grid/true}

		Turns the central lines back on after a |grid = false| (for example, a global one set with |\pgfgoset|).
	\end{stylekey}

	\begin{stylekey}{/pgfgo/goban/grid/false}
		
		Disables the central lines, leaving only the border and stones visible.
		
		\begin{codeexample}[]
			\begin{goban}[3]
				\stone[b]{A1}
				\stone[w]{B2}
			\end{goban}
			
			\begin{goban}[3, grid = false]
				\stone[b]{A1}
				\stone[w]{B2}
			\end{goban}
		\end{codeexample}
		
		The first board shows the default grid.
	\end{stylekey}
	
	\begin{key}{/pgfgo/goban/grid/line=\meta{options} (default true)}
		Allows controlling the grid lines.
	\end{key}

	\begin{stylekey}{/pgfgo/goban/grid/line/true}
		Draws the lines. It is the initial value.
	\end{stylekey}

	\begin{stylekey}{/pgfgo/goban/grid/line/false}
		Does not draw the lines, but draws the star points (unlike |grid = false|).

		\begin{codeexample}[]
			\begin{goban}[9, grid = {line = false}, hoshi = {radius = 2pt}]
				\stones[b]{C3, E5}
			\end{goban}
		\end{codeexample}
	\end{stylekey}
	
	\begin{key}{/pgfgo/goban/grid/line/color =\meta{color} (initially black)}
		
		Allows changing the color of the grid lines.
		
		\begin{codeexample}[]
			\pgfgoset{goban = {5, partial = B2:D4}}
			
			\begin{goban}[grid = {line = {color = blue}}, name = a]
				\stones[b]{C2,D3}
				\stone[w]{B4}
			\end{goban}
			
			\begin{goban}[grid = {line = {color = red}}, resume = a]
			\end{goban}
		\end{codeexample}
	\end{key}
	
	\begin{key}{/pgfgo/goban/grid/line/opacity =\meta{float} (initially 1)}
		
		Sets the opacity of the grid lines, from 0 (transparent) to 1 (opaque).
		
		\begin{codeexample}[]
			\pgfgoset{goban = {5, partial = B2:D4}}
			
			\begin{goban}[name = a]
				\stones[b]{C2,D3}
				\stone[w]{B4}
			\end{goban}
			
			\begin{goban}[grid = {line = {opacity = 0.2}}, resume = a]
			\end{goban}
		\end{codeexample}
	\end{key}
	
	\begin{key}{/pgfgo/goban/grid/line/width =\meta{dimension} (initially 0.55pt)}
		
		Defines the thickness of the grid lines.
		
		\begin{codeexample}[]
			\pgfgoset{goban = {5, partial = B2:D4}}
			
			\begin{goban}[name = a]
				\stones[b]{C2,D3}
				\stone[w]{B4}
			\end{goban}
			
			\begin{goban}[grid = {line = {width = 1.5pt}}, resume = a]
			\end{goban}
		\end{codeexample}
	\end{key}
	
	\begin{key}{/pgfgo/goban/grid/x sep =\meta{dimension} (initially 1.25em)}
		Sets the spacing between vertical lines, adjusting the width of the board’s columns.
	\end{key}
	
	\begin{key}{/pgfgo/goban/grid/y sep =\meta{dimension} (initially 1.25em)}
		Sets the spacing between horizontal lines, adjusting the height of the rows.
	\end{key}
	
	\begin{key}{/pgfgo/goban/grid/sep =\meta{dimension} (initially 1.25em)}
		Simultaneously defines the vertical and horizontal spacing of the lines, ensuring uniform spacing in both directions. This is a convenient option for maintaining balanced proportions.
	\end{key}

	\begin{key}{/pgfgo/goban/grid/image =\meta{options}}
		Draws an image instead of the lines of the grid, with the opacity of |grid/line/opacity|. With |partial|, the corresponding part is shown. The star points are still drawn (see |hoshi|).

		\begin{codeexample}[]
			\begin{goban}[9, grid = {image = {file = grid.png}}, hoshi = false]
				\stones[b]{C3}\stone[w]{D4}
			\end{goban}

			\begin{goban}[9, grid = {image = {file = grid.png}}, hoshi = false, partial = C3:G7]
			\end{goban}
		\end{codeexample}
	\end{key}

	\begin{key}{/pgfgo/goban/grid/image/file =\meta{file}}
		The image file.
	\end{key}

	\begin{key}{/pgfgo/goban/grid/image/fit =\meta{stretch or tile} (initially stretch)}
		With |stretch|, the image is stretched from A1 to the opposite corner of the whole board, so the outer lines must be on the edges of the image. With |tile|, it is drawn once on each square of the grid (between four intersections), stretched to the size of the square: the image of one square builds the whole board.

		\begin{codeexample}[]
			\begin{goban}[9, grid = {image = {file = cell.png, fit = tile}}, hoshi = false]
				\stones[b]{C3}\stone[w]{D4}
			\end{goban}
		\end{codeexample}
	\end{key}

	\begin{key}{/pgfgo/goban/grid/image/tiles =\meta{columns}|x|\meta{rows} (initially one per square)}
		With |fit = tile|, the mesh in which the image is repeated, over the area from A1 to the opposite corner of the whole board; each image is stretched to the size of its cell. A single number is used for both directions. Without this key, there is one image per square.

		\begin{codeexample}[]
			\begin{goban}[9, grid = {image = {file = cell.png, fit = tile, tiles = 4}}, hoshi = false]
			\end{goban}
		\end{codeexample}
	\end{key}

	\begin{key}{/pgfgo/goban/grid/image/rotate =\meta{angle}, |random| or \meta{min}|:|\meta{max} (initially 0)}
		Turns the image, like |background/fill/image/rotate|: in quarter turns and, with |random| or a range, each tile with its own.
	\end{key}

	\begin{stylekey}{/pgfgo/goban/grid/image/false}
		Removes the image: the lines come back.
	\end{stylekey}

	\subsection{Star Points}
	\label{subsec:hoshi}

	The star points (\emph{hoshi}) are drawn on the grid, in its color. By default they are the usual ones: on each side, the fourth line from the edge (the third on boards with fewer than 13 lines); the center if the size is odd; and, from 15 lines on, the middle of the sides. A 19 board has nine, and a 13 or 9 board, five. Boards with fewer than 7 lines have none. Stones cover them.

	\begin{key}{/pgfgo/goban/hoshi=\meta{options} (default true)}
		Options of the star points.

		\begin{codeexample}[]
			\begin{goban}[9]
				\stones[b]{C3, E5}
			\end{goban}

			\begin{goban}[9, hoshi = false]
				\stones[b]{C3, E5}
			\end{goban}
		\end{codeexample}
	\end{key}

	\begin{stylekey}{/pgfgo/goban/hoshi/true}
		Draws the star points. It is the initial value.
	\end{stylekey}

	\begin{stylekey}{/pgfgo/goban/hoshi/false}
		Does not draw the star points.
	\end{stylekey}

	\begin{key}{/pgfgo/goban/hoshi/radius=\meta{dimension} (initially 1.5pt)}
		Radius of the points.
	\end{key}

	\begin{key}{/pgfgo/goban/hoshi/color=\meta{color}}
		Color of the points. Without this \emph{key}, they take the color of the grid.

		\begin{codeexample}[]
			\begin{goban}[9, hoshi = {radius = 2.5pt, color = red}]
			\end{goban}
		\end{codeexample}
	\end{key}

	\begin{key}{/pgfgo/goban/hoshi/opacity=\meta{float}}
		Opacity of the points (also of their image). Without this \emph{key}, they take the opacity of the grid.
	\end{key}

	\begin{key}{/pgfgo/goban/hoshi/at=\meta{coordinates}}
		Chooses the points, with the same syntax as |\stones|. Empty, the usual ones are used.

		\begin{codeexample}[]
			\begin{goban}[9, hoshi = {at = {C3, G7}}]
			\end{goban}
		\end{codeexample}
	\end{key}
	
	\begin{key}{/pgfgo/goban/hoshi/image=\meta{options}}
		Draws an image on each star point instead of the circle.

		\begin{codeexample}[]
			\begin{goban}[9, hoshi = {image = {file = hoshi.png}, radius = 3pt}]
			\end{goban}
		\end{codeexample}
	\end{key}

	\begin{key}{/pgfgo/goban/hoshi/image/file=\meta{file}}
		The image file.
	\end{key}

	\begin{key}{/pgfgo/goban/hoshi/image/width=\meta{dimension} (initially $2 \times$ |radius|)}
		The width of the image; the height keeps the proportions.
	\end{key}

	\begin{key}{/pgfgo/goban/hoshi/image/rotate=\meta{angle}, |random| or \meta{min}|:|\meta{max} (initially 0)}
		Turns the image, like |stone/image/rotate| (p.~\pageref{subsec:stoneimages}): with |random| or a range, each point gets a different angle, always the same.
	\end{key}

	\begin{stylekey}{/pgfgo/goban/hoshi/image/false}
		Removes the image: the circles come back.
	\end{stylekey}

	\subsection{Border}
	
	The board’s border is the outline that frames the grid and stones, providing a clear visual boundary. By default, it is active, has a white background, and a black outline.
	
	\begin{key}{/pgfgo/goban/background=\meta{options} (default true)}
		Controls the properties of the board’s border, such as its visibility, color, thickness, and shape, allowing detailed customization.
	\end{key}
	
	\begin{stylekey}{/pgfgo/goban/background/true}

		Turns the board’s background and border back on after a |background = false|.
	\end{stylekey}

	\begin{stylekey}{/pgfgo/goban/background/false}
		
		Completely disables the board’s border, removing both the outline and fill. This is useful for minimalist diagrams or when the document’s background already provides sufficient contrast.
		
		\begin{codeexample}[]
			\pgfgoset{goban = {5, partial = B2:D4}}
			
			\begin{goban}[name = a]
				\stones[b]{C2,D3}
				\stone[w]{B4}
			\end{goban}
			
			\begin{goban}[background = false, resume = a]
			\end{goban}
		\end{codeexample}
	\end{stylekey}
	
	\begin{key}{/pgfgo/goban/background/fill=\meta{options} (default true)}
		Controls options related to the board’s background fill.
	\end{key}
	
	\begin{key}{/pgfgo/goban/background/fill=\meta{boolean} (initially true)}
		
		Determines whether the board’s background fill is active. If disabled, the area inside the border will be transparent, showing only the outline (if enabled).
		
		\begin{codeexample}[]
			\pgfgoset{goban = {5, partial = B2:D4}}
			
			\begin{goban}[background = {line = {color = red}}, name = a]
				\stones[b]{C2,D3}
				\stone[w]{B4}
			\end{goban}
			
			\begin{goban}[background = {line = false}, resume = a]
			\end{goban}
		\end{codeexample}
	\end{key}
	
	\begin{key}{/pgfgo/goban/background/fill/color =\meta{color} (initially white)}
		
		Defines the fill color of the board.
		
		\begin{codeexample}[]
			\pgfgoset{goban = {5, partial = B2:D4}}
			
			\begin{goban}[background = {fill = {color = orange!40}}, name = a]
				\stones[b]{C2,D3}
				\stone[w]{B4}
			\end{goban}
			
			\begin{goban}[background = {fill = {color = olive!30}}, resume = a]
			\end{goban}
		\end{codeexample}
	\end{key}
	
	\begin{key}{/pgfgo/goban/background/fill/opacity =\meta{float} (initially 1)}
		
		Controls the opacity of the fill, from 0 (transparent) to 1 (opaque).
		
		\begin{codeexample}[]
			\pgfgoset{goban = {5, partial = B2:D4}}
			
			\begin{goban}[background = {fill = {color = orange!40}}, name = a]
				\stones[b]{C2,D3}
				\stone[w]{B4}
			\end{goban}
			
			\begin{goban}[background = {fill = {color = orange!40}, fill = {opacity = 0.5}}, resume = a]
			\end{goban}
		\end{codeexample}
		
	\end{key}
	
	\begin{key}{/pgfgo/goban/background/fill/image =\meta{options}}
		A background image (for example, the wood of a board), over the fill color and with its opacity. The image is placed according to the whole board: with |partial|, the corresponding part is shown, clipped by the border and its rounded corners. Marks on empty points, which cover the grid with the color of the background, show the part of the image under them, and so does the |dim| mark. The halo of arrows and lines keeps the color of the background.

		\begin{codeexample}[]
			\begin{goban}[9, background = {fill = {image = {file = wood.jpg}}}]
				\stones[b]{C3}\stone[w]{D4}
				\marks[circle]{E5, C3}
			\end{goban}

			\begin{goban}[9, background = {fill = {image = {file = wood.jpg}}, rounded corners = 6pt}, partial = A1:E5]
				\marks[triangle]{C3}
			\end{goban}
		\end{codeexample}
	\end{key}

	\begin{key}{/pgfgo/goban/background/fill/image/file =\meta{file}}
		The file of the background image.
	\end{key}

	\begin{key}{/pgfgo/goban/background/fill/image/fit =\meta{stretch, cover or tile} (initially cover)}
		How the image fills the background of the whole board. |stretch| stretches it to fill the whole background, without keeping the proportions. |cover| scales it, keeping the proportions, until it covers the background, centered: what is left over is not shown. |tile| repeats it as tiles, from a corner, a side or the center of the whole board (see |anchor|).

		\begin{codeexample}[]
			\begin{goban}[9, background = {fill = {image = {file = tile.png, fit = tile, width = 1cm}}}]
			\end{goban}

			\begin{goban}[9, background = {fill = {image = {file = tile.png, fit = tile, width = 1cm}}}, partial = D4:H8]
			\end{goban}
		\end{codeexample}
	\end{key}

	\begin{key}{/pgfgo/goban/background/fill/image/anchor =\meta{position} (initially south west)}
		With |fit = tile|, where the tiles start, on the background of the whole board. With a corner (|south west|, |south east|, |north west| or |north east|), that corner gets a whole tile. With a side (|north|, |south|, |east| or |west|), the first tile touches that side, centered along it. With |center|, a tile is in the center of the board.

		\begin{codeexample}[]
			\begin{goban}[5, background = {fill = {image = {file = tile.png, fit = tile, width = 1cm, anchor = north east}}}]
			\end{goban}

			\begin{goban}[5, background = {fill = {image = {file = tile.png, fit = tile, width = 1cm, anchor = north}}}]
			\end{goban}
		\end{codeexample}
	\end{key}

	\begin{key}{/pgfgo/goban/background/fill/image/width =\meta{dimension} (initially the natural width)}
		With |fit = tile|, the width of each tile; the height keeps the proportions.
	\end{key}

	\begin{key}{/pgfgo/goban/background/fill/image/tiles =\meta{columns}|x|\meta{rows}}
		With |fit = tile|, instead of tiles of width |width| from |anchor|, splits the background of the whole board into that mesh and puts the image on each cell, stretched to its size. A single number is used for both directions.

		\begin{codeexample}[]
			\begin{goban}[9, background = {fill = {image = {file = cell.png, fit = tile, tiles = 3x3}}}]
			\end{goban}
		\end{codeexample}
	\end{key}

	\begin{key}{/pgfgo/goban/background/fill/image/rotate =\meta{angle}, |random| or \meta{min}|:|\meta{max} (initially 0)}
		Turns the image in quarter turns: the angle is rounded to 0, 90, 180 or 270 degrees, so that the turned image still fills its place without gaps. With 90 or 270, the width and the height are swapped: with |cover| the image keeps its proportions, but a tile of width |width| that is not square is stretched. With |random| or a range, as in |stone/image/rotate| (p.~\pageref{subsec:stoneimages}), each tile has its own turn, always the same: the repetition shows less.

		\begin{codeexample}[]
			\begin{goban}[5, background = {fill = {image = {file = wood.jpg, rotate = 90}}}]
			\end{goban}

			\begin{goban}[5, background = {fill = {image = {file = wood.jpg, fit = tile, width = 1cm, rotate = random}}}]
			\end{goban}
		\end{codeexample}
	\end{key}

	\begin{stylekey}{/pgfgo/goban/background/fill/image/false}
		Removes the background image.
	\end{stylekey}

	\begin{key}{/pgfgo/goban/background/line=\meta{options} (default true)}
		Controls the options for the background’s outline.
	\end{key}
	
	\begin{key}{/pgfgo/goban/background/line=\meta{boolean} (initially true)}
		Enables or disables the border line.
		
		\begin{codeexample}[]
			\pgfgoset{goban = {5, partial = B2:D4}}
			
			\begin{goban}[background = {line = {color = red}}, name = a]
				\stones[b]{C2,D3}
				\stone[w]{B4}
			\end{goban}
			
			\begin{goban}[background = {line = false}, resume = a]
			\end{goban}
		\end{codeexample}
	\end{key}
	
	\begin{key}{/pgfgo/goban/background/line/color =\meta{color} (initially black)}
		
		Defines the color of the border line, allowing it to stand out from the background or grid lines.
		
		\begin{codeexample}[]
			\pgfgoset{goban = {5, partial = B2:D4}}
			
			\begin{goban}[background = {line = {color = red}}, name = a]
				\stones[b]{C2,D3}
				\stone[w]{B4}
			\end{goban}
			
			\begin{goban}[background = {line = {color = blue}}, resume = a]
			\end{goban}
		\end{codeexample}
	\end{key}
	
	\begin{key}{/pgfgo/goban/background/line/opacity =\meta{float} (initially 1)}
		
		Adjusts the opacity of the border line.
		
		\begin{codeexample}[]
			\pgfgoset{goban = {5, partial = B2:D4}}
			
			\begin{goban}[background = {line = {color = red}}, name = a]
				\stones[b]{C2,D3}
				\stone[w]{B4}
			\end{goban}

	\subsection{Partial Board}
	
	In many cases, it’s not necessary to represent a full board, but rather just a rectangular section—whether due to space constraints, to focus attention on a specific area, or to simplify diagram creation. The |goban| environment includes the \emph{key} |partial|, which defines this region with two opposite vertices of a rectangle, regardless of the order in which they are given. For example, |partial = A1:D3| is equivalent to |partial = D3:A1|.
	
	\begin{figure}[H]
		\centering
		
		\begin{tikzpicture}
			\node[anchor=north] (X) {
				\begin{goban}[label = {at = {north, west}}]
					\coordinate (A) at (4,6);
					\coordinate (B) at (9,9);
					\coordinate (C) at (A|-B);
					\coordinate (D) at (A-|B);
					
					\tikzset{punto/.style = {circle, fill = #1, draw, inner sep=0.25em, fill opacity=0.5, anchor=center}}
					
					\node[punto = red] at (A){};
					\node[punto = red] at (B){};
					\node[punto = blue] at (C){};
					\node[punto = blue] at (D){};
					
					\draw[red, {latex}-{latex}, shorten <=5pt, shorten >= 5pt] (A)--(B);
					\draw[blue, {latex}-{latex}, shorten <=5pt, shorten >= 5pt] (C)--(D);
					
					\node[fit=(A)(B), draw, inner sep=0.6em, rounded corners, line width=1pt, fill opacity=0.1, fill=orange] {};
				\end{goban}
			};
			
			\node[anchor=north, xshift=15em] {
				\tikz{
					\node[inner sep=0] (Y) {
						\begin{goban}[label = {at = {north, west}}, partial = I9:D6]
						\end{goban}
					};

						{\color{blue}|partial = D9:I6|}\\

			};
		\end{tikzpicture}
	\end{figure}
	
	\begin{key}{/pgfgo/goban/partial=\meta{intersection}|:|\meta{intersection} or |false|}
		Shows only a part of the board: the rectangle that has those two intersections as opposite corners, in any order. It can also be written |partial = {B2, D4}| or |partial = {B2 to D4}|. With |false|, it shows the whole board again (for example, after a global |partial| set with |\pgfgoset|).
	\end{key}
	
	\begin{codeexample}[]
		\tikzset{flecha/.style = {-{latex[]}, line width = 1pt, red}}
		
		\begin{goban}[partial = A1:D3]
			\draw[flecha] (1,1)--(4,3); % From A1 to D3
		\end{goban}
		
		\begin{goban}[partial = A9:I6]
			\draw[flecha] (1,9)--(9,6); % From A9 to I6
		\end{goban}
		
		\begin{goban}[partial = E2:B4]
			\draw[flecha] (5,2)--(2,4); % From E2 to B4
		\end{goban}
	\end{codeexample}
	
	\subsection{Board Labels}
	
	Labels are the alphanumeric indicators (letters and numbers) that appear on the sides of the board to identify intersections.
	
	\begin{key}{/pgfgo/goban/label=\meta{options} (default true)}
		Controls the properties of the labels, such as their position and format.
	\end{key}
	
	\begin{key}{/pgfgo/goban/label/true}
		Puts the letters below and the numbers on the left: |A1| is read from the lower left corner. It is the same as |label = {at = {south, west}}|; for the four sides, |label = {at = all}|.

		\begin{codeexample}[]
			\begin{goban}[5, label]
				\stones[b]{A1}
			\end{goban}
		\end{codeexample}
	\end{key}

	\begin{key}{/pgfgo/goban/label/false}
		Disables all labels.
	\end{key}
	
	\begin{key}{/pgfgo/goban/label/at =\meta{options}}
		
		Defines on which sides of the board the labels appear. Options include combinations of cardinal positions or predefined values like "all" or "none".
		
		\begin{stylekey}{/pgfgo/goban/label/at/all}
			
			Displays labels on all sides of the board (north, south, east, and west).
		\end{stylekey}
		
		\begin{stylekey}{/pgfgo/goban/label/at/none}
			
			Equivalent to |label=false|, removing all labels.
		\end{stylekey}
		
		\begin{stylekey}{/pgfgo/goban/label/at/east}
			Places labels only on the right side (vertical numbers).
		\end{stylekey}
		\begin{stylekey}{/pgfgo/goban/label/at/right}
			Equivalent to |east|.
		\end{stylekey}
		
		\begin{stylekey}{/pgfgo/goban/label/at/north}
			Places labels only at the top (horizontal letters).
		\end{stylekey}
		\begin{stylekey}{/pgfgo/goban/label/at/above}
			Equivalent to |north|.
		\end{stylekey}
		
		\begin{stylekey}{/pgfgo/goban/label/at/south}
			Places labels only at the bottom (horizontal letters).
		\end{stylekey}
		\begin{stylekey}{/pgfgo/goban/label/at/below}
			Equivalent to |south|.
		\end{stylekey}
		
		\begin{stylekey}{/pgfgo/goban/label/at/west}
			Places labels only on the left side (vertical numbers).
		\end{stylekey}
		\begin{stylekey}{/pgfgo/goban/label/at/left}
			Equivalent to |west|.
		\end{stylekey}
		
		\begin{codeexample}[]
			\begin{goban}[12, label = {at = {all}}]
			\end{goban}
			
			\begin{goban}[partial = A6:I9, label = {at = north}]
			\end{goban}
			
			\begin{goban}[partial = B2:E4, label = {at = {north, east}}]
			\end{goban}
		\end{codeexample}
	\end{key}
	
	\begin{key}{/pgfgo/goban/label/font size = \meta{dimension} (initially the current font size)}

		Adjusts the font size of the labels.

		\begin{codeexample}[]
			\begin{goban}[5, label = {at = {south, west}}]
			\end{goban}

			\begin{goban}[5, label = {at = {south, west}, font size = 6pt}]
			\end{goban}
		\end{codeexample}
	\end{key}

	\begin{key}{/pgfgo/goban/label/font = \meta{font commands} (initially \texttt{\string\sffamily})}

		Sets the font of the labels.

		\begin{codeexample}[]
			\begin{goban}[5, label = {at = {south, west}, font = \bfseries}]
			\end{goban}
		\end{codeexample}
	\end{key}
	
	\begin{key}{/pgfgo/goban/label/sep = \meta{dimension} (initially 0.35cm)}
		
		Controls the distance between the labels and the board’s border.
	\end{key}
	
	\begin{key}{/pgfgo/goban/label/color = \meta{color} (initially black)}
		
		Defines the text color of the labels.
	\end{key}

	\begin{key}{/pgfgo/goban/label/columns = \meta{letters, numbers or kanji} (initially letters)}
		How the coordinates of the columns (above and below) are written: with letters (as in the commands, without the I if |use i = false|), with numbers, or with numbers in kanji ({\pgfgokanji 一, 二, …, 十, 十一}, …).
	\end{key}

	\begin{key}{/pgfgo/goban/label/rows = \meta{letters, numbers or kanji} (initially numbers)}
		The same for the rows (at the sides). The kanji of the rows are aligned on the side of the board, so that the two-character ones do not overlap it.
	\end{key}

	\begin{key}{/pgfgo/goban/label/columns direction = \meta{left to right or right to left} (initially left to right)}
		The side from which the columns are counted.
	\end{key}

	\begin{key}{/pgfgo/goban/label/rows direction = \meta{bottom to top or top to bottom} (initially bottom to top)}
		The side from which the rows are counted. These four options only change what is written on the border: the coordinates of the commands (|C3|) stay the same. On a partial board, they are counted from the edge of the whole board.

		\begin{codeexample}[]
			\begin{goban}[5, label = {at = {north, west}, columns = numbers, rows direction = top to bottom}]
				\stones[b]{C4}
			\end{goban}
		\end{codeexample}
	\end{key}

	\begin{key}{/pgfgo/goban/label/style = \meta{western, japanese or chinese} (initially western)}
		Sets the four options above as usual in each place.

		\begin{stylekey}{/pgfgo/goban/label/style/western}
			Columns with letters from the left and rows with numbers from the bottom.
		\end{stylekey}

		\begin{stylekey}{/pgfgo/goban/label/style/japanese}
			Columns with numbers from the right and rows with kanji from the top, as in Japanese books: the top right corner is 1-{\pgfgokanji 一}.
		\end{stylekey}

		\begin{stylekey}{/pgfgo/goban/label/style/chinese}
			Columns with numbers from the left and rows with kanji from the top.
		\end{stylekey}

		The kanji need a font that has them. With Lua\LaTeX, for example, IPAex (included in \TeX\ Live and MiK\TeX):

		\begin{codeexample}[]
			\font\kanji = {file:ipaexm.ttf} at 8pt
			\begin{goban}[9, label = {at = {north, east}, style = japanese, font = \kanji}]
				\stones[b]{G7}
			\end{goban}
		\end{codeexample}
	\end{key}
	
	\subsection{Captures}
	\label{subsec:captures}

	With the Lua kernel (p.~\pageref{subsec:kernel}), \pgfgoname\ can keep track of the stones on the board and remove the ones left without liberties, as in a game.

	\begin{key}{/pgfgo/goban/captures=\meta{remove, fade or false} (default remove, initially false)}

		Turns captures on. When a move leaves an opposing group without liberties, that group is removed from the board. If it leaves its own group without liberties (a suicide), that group is removed. Stones placed with |\stone| and |\stones| also count; neutral stones and prisoners do not. A move outside a partial board also captures. A move on an occupied point produces a warning, because it is not possible in a game. With |kernel = expl3|, this \emph{key} raises an error.

		\begin{codeexample}[]
			\pgfgoset{goban = {5, partial = A1:C3}}

			\begin{goban}
				\moves*{B1,A1,A2}
			\end{goban}

			\begin{goban}[captures]
				\moves*{B1,A1,A2} % 3 captures 2.
			\end{goban}
		\end{codeexample}

		Since \pgfname\ cannot erase what it has already drawn, with |captures| the stones are drawn when the |goban| ends. For this reason, if the style is changed with |\pgfgoset| inside a |goban| with captures, the change does not reach the stones; the options of each command do.
	\end{key}

	\begin{stylekey}{/pgfgo/goban/captures/remove}

		Captured stones disappear. It is the same as |captures| or |captures = true|.
	\end{stylekey}

	\begin{stylekey}{/pgfgo/goban/captures/fade}

		Captured stones stay on the board, drawn as prisoners (with the style of |\prisoner| and |/pgfgo/prisoner|).

		\begin{codeexample}[]
			\pgfgoset{goban = {5, partial = A1:E3}}

			\begin{goban}[captures = fade]
				\stones[w]{C1,D1}
				\stones[b]{B1,C2,D2}
				\moves*{E1}
			\end{goban}
		\end{codeexample}
	\end{stylekey}

	\begin{stylekey}{/pgfgo/goban/captures/false}

		Turns captures off. It is the initial value.
	\end{stylekey}

	\begin{key}{/pgfgo/goban/captured marks=\meta{keep or remove} (initially keep)}

		Decides what happens to the marks on a stone when it is captured. It only works with |captures|. Marks on empty points do not change.

		\begin{codeexample}[]
			\pgfgoset{goban = {5, partial = A1:C3, captures}}

			\begin{goban}
				\stones[w]{A1}\stones[b]{B1}
				\marks[triangle]{A1}
				\moves*{A2}
			\end{goban}

			\begin{goban}[captured marks = remove]
				\stones[w]{A1}\stones[b]{B1}
				\marks[triangle]{A1}
				\moves*{A2} % the triangle goes with the white stone.
			\end{goban}
		\end{codeexample}
	\end{key}

	\begin{stylekey}{/pgfgo/goban/captured marks/keep}

		The marks stay on the empty point. It is the initial value.
	\end{stylekey}

	\begin{stylekey}{/pgfgo/goban/captured marks/remove}

		The marks go with the stone, also with |captures = fade|. Sequences keep their numbers.
	\end{stylekey}

	\begin{command}{\pgfgoprisoners\marg{player}}
		Writes how many prisoners the player (|black| or |b|, |white| or |w|) has in the last |goban| with |captures|: the opponent’s stones that their moves captured.

		\begin{codeexample}[]
			\pgfgoset{goban = {5, partial = A1:C3}}

			\begin{goban}[captures]
				\moves*{B1,A1,A2}
			\end{goban}

			Black: \pgfgoprisoners{black}. White: \pgfgoprisoners{white}.
		\end{codeexample}
	\end{command}

	\subsection{Territory}
	\label{subsec:territory}

	At the end of the game, the territory can be marked by hand with |\marks| (p.~\pageref{sec:marks}) or with |\territory|, which also counts it.

	\begin{command}{\territory\opt{\oarg{options}}\marg{polygon}}
		Marks as territory the points of the polygon with those vertices (in order), including its border. It does not mark the points that have a living stone: prisoners count as territory. Vertices are written as in |\stones|; a single one marks a point, and two, a segment. It works with both kernels.

		\begin{codeexample}[width=4.75cm]
			\begin{goban}[label = {at = {above, left}}]
		\stones[b] {A6|-B5|-D1|-E2, C4, E9-|F4, D5-|G7--D7, H6, H8, I7}
		\stones[w] {E8-|C5, B:E6, A7, B8, D9, I6|- H5, H2:3, F1|-C3, C2, D:E4, G4}
		\prisoners[b]{H1-|I2}
		\prisoner[w] {A1}

				\territory[b]{A1,A5,B5,B1}
				\territory[b]{F9,F7,G7,G6,I6,I9}
				\territory[w]{F1,F4,G4,G5,I5,I1}
				\territory[w]{A9,A6,C6,C8,D8,D9}
			\end{goban}
		\end{codeexample}
	\end{command}

	\begin{command}{\territory\opt{\oarg{options}}\marg{auto}}
		With |auto| instead of a polygon, it finds the territory by itself: each region of points without living stones belongs to the color around it. If both colors touch it (dame, seki), or none, it is neutral. Prisoners count as dead stones. Without a player, it marks the territory of both; with |b| or |w|, only that player's. Only with the Lua kernel (p.~\pageref{subsec:kernel}).

		\begin{codeexample}[width=4.75cm]
			\begin{goban}[label = {at = {above, left}}]
		\stones[b] {A6|-B5|-D1|-E2, C4, E9-|F4, D5-|G7--D7, H6, H8, I7}
		\stones[w] {E8-|C5, B:E6, A7, B8, D9, I6|- H5, H2:3, F1|-C3, C2, D:E4, G4}
		\prisoners[b]{H1-|I2}
		\prisoner[w] {A1}

				\territory[dame]{auto}
			\end{goban}
		\end{codeexample}
	\end{command}

	\begin{key}{/pgfgo/territory/black}
		The territory belongs to Black. It can also be written |b|.
	\end{key}

	\begin{key}{/pgfgo/territory/white}
		The territory belongs to White. It can also be written |w|.
	\end{key}

	\begin{key}{/pgfgo/territory/neutral}
		Neutral points. It is what a polygon without a player marks. It can also be written |n|.
	\end{key}

	\begin{key}{/pgfgo/territory/b}
		Short for |black|.
	\end{key}

	\begin{key}{/pgfgo/territory/w}
		Short for |white|.
	\end{key}

	\begin{key}{/pgfgo/territory/n}
		Short for |neutral|.
	\end{key}

	\begin{key}{/pgfgo/territory/draw=\meta{marks, fill or region} (initially marks)}
		With |marks|, every point gets the territory mark of the player. With |fill|, the region is filled below the stones.

		\begin{codeexample}[width=4.75cm]
			\begin{goban}[label = {at = {above, left}}]
		\stones[b] {A6|-B5|-D1|-E2, C4, E9-|F4, D5-|G7--D7, H6, H8, I7}
		\stones[w] {E8-|C5, B:E6, A7, B8, D9, I6|- H5, H2:3, F1|-C3, C2, D:E4, G4}
		\prisoners[b]{H1-|I2}
		\prisoner[w] {A1}

				\territory[draw = fill]{auto}
			\end{goban}
		\end{codeexample}

		With |region|, the territory of each color is a single polygon: every square of the grid with a corner in the territory is painted (dead stones count as territory). Its sides run along the lines of the board, its vertices are always on intersections, and it reaches the edge of the board. Two areas that touch at a single corner are kept apart. All the |\territory| of a color in the same |goban| (also the |TB| and |TW| territory of |\sgf|) make a single region, which is drawn at the end of the |goban| with the style of the last one. Neutral points are filled one by one, as with |fill|. The outline and its vertices can be drawn on top, with |line| and |vertices|, and |\pgfgoregion| gives its properties.

		\begin{codeexample}[width=4.75cm]
			\begin{goban}[label = {at = {above, left}}]
		\stones[b] {A6|-B5|-D1|-E2, C4, E9-|F4, D5-|G7--D7, H6, H8, I7}
		\stones[w] {E8-|C5, B:E6, A7, B8, D9, I6|- H5, H2:3, F1|-C3, C2, D:E4, G4}
		\prisoners[b]{H1-|I2}
		\prisoner[w] {A1}

				\territory[draw = region]{auto}
			\end{goban}
		\end{codeexample}
	\end{key}

	\begin{key}{/pgfgo/territory/dame=\meta{boolean} (default true, initially false)}
		With |auto|, it also marks the neutral points.
	\end{key}

	\begin{key}{/pgfgo/territory/forget}
		|resume| does not bring this territory, even with |territory| (p.~\pageref{subsec:remember}).
	\end{key}

	\begin{key}{/pgfgo/players/black/territory=\meta{territory options}}
		Style of the territory of each player (|black|, |white| or |neutral|).
	\end{key}

	\begin{key}{/pgfgo/players/black/territory/mark=\meta{mark options} (initially t-circle)}
		Mark of every point, with |draw = marks|. It is |t-circle| for Black and |t-square| for White and the neutral points.
	\end{key}

	\begin{key}{/pgfgo/players/black/territory/fill=\meta{fill options} (default true)}
		Fill, with |draw = fill| and |draw = region|.
	\end{key}

	\begin{key}{/pgfgo/players/black/territory/fill/true}
		Draws the fill. This is the initial value.
	\end{key}

	\begin{key}{/pgfgo/players/black/territory/fill/false}
		Does not draw the fill: with |draw = region|, only the outline or the vertices are left.
	\end{key}

	\begin{key}{/pgfgo/players/black/territory/fill/color=\meta{color} (initially black)}
		Color of the fill: |black| for Black, |white!80!black| for White and |gray| for the neutral points.
	\end{key}

	\begin{key}{/pgfgo/players/black/territory/fill/opacity=\meta{number} (initially 0.35)}
		Opacity of the fill: |0.35| for Black and the neutral points, |0.7| for White.
	\end{key}

	\begin{key}{/pgfgo/players/black/territory/line=\meta{line options} (default true)}
		Outline of the region, with |draw = region|. Alone or with |true|, it draws the outline; options alone do not turn it on.

		\begin{codeexample}[width=4.75cm]
			\begin{goban}[label = {at = {above, left}}]
		\stones[b] {A6|-B5|-D1|-E2, C4, E9-|F4, D5-|G7--D7, H6, H8, I7}
		\stones[w] {E8-|C5, B:E6, A7, B8, D9, I6|- H5, H2:3, F1|-C3, C2, D:E4, G4}
		\prisoners[b]{H1-|I2}
		\prisoner[w] {A1}

				\territory[b, draw = region, line,
					vertices]{auto}
				\territory[w, draw = region,
					line = {true, color = red}]{auto}
			\end{goban}
		\end{codeexample}
	\end{key}

	\begin{key}{/pgfgo/players/black/territory/line/true}
		Draws the outline.
	\end{key}

		Does not draw the outline. This is the initial value.
	\end{key}

		Color of the outline: |black| for Black and |white!40!black| for White.

		Width of the outline.

	\begin{key}{/pgfgo/players/black/territory/line/opacity=\meta{number} (initially 1)}
		Opacity of the outline and of the vertices.

	\end{key}

		Dots on the vertices of the outline, with |draw = region|, like the star points. They are drawn on the same layer as the outline. Alone or with |true|, it draws them; options alone do not turn them on.
	\end{key}

		Radius of the dots.
	\end{key}

		Color of the dots. Without this key, they take the color of the outline.
	\end{key}

	\begin{command}{\pgfgoscore\opt{\oarg{options}}\marg{player}}
		Writes the score of the player (|black| or |b|, |white| or |w|) in the last |goban|, with Japanese rules: the points of their territory, plus the opponent's dead stones in it, plus the stones they captured (with |captures|), plus the |komi| for White. A point marked twice counts once.

		\begin{codeexample}[code only]
			Black: \pgfgoscore{black}. White: \pgfgoscore[komi = 6.5]{white}.
		\end{codeexample}

		With the board above, it writes "Black: 11. White: 23.5".
	\end{command}

	\begin{key}{/pgfgo/score/komi=\meta{number} (initially 0)}
		Komi added to the score of White. It can be set for the whole document with |\pgfgoset{score = {komi = 6.5}}|.
	\end{key}

	\begin{command}{\pgfgoregion\marg{player}\marg{property}}
		Writes a property of the region (|draw = region|) of the player (|black| or |b|, |white| or |w|) in the current |goban| or in the last one. It is expandable, so it can be used in a computation or inside the coordinates of |\marks|. The properties are those of the whole outline, even on a partial board. Without a region, the numbers are 0. The properties are:
		\begin{itemize}
			\item |polygons|: the number of outlines, outer ones and holes.
			\item |outlines|: the number of outer outlines, one for each separate area.
			\item |holes|: the number of holes.
			\item |vertices|: the number of vertices of all the outlines.
			\item |area|: the number of painted squares.
			\item |perimeter|: the length of all the outlines, in squares.
			\item |polygon| \meta{i}: the vertices of outline \meta{i}, as coordinates separated by commas (for example, |A1,D1,D4,A4|). Empty if there are not that many outlines.
		\end{itemize}
		Each outline starts at its lowest vertex and, among those, at the leftmost one. Outer outlines go counterclockwise and holes, clockwise. The outlines are sorted by that first vertex, from bottom to top.

		\begin{codeexample}[width=4.75cm]
			\begin{goban}[label = {at = {above, left}}]
				\stones[w]{D4 R F6}
				\territory[b, draw = region, line]{B2,B8,H8,H2}
				\marks[triangle]{\pgfgoregion{b}{polygon 2}}
			\end{goban}

			Outlines: \pgfgoregion{b}{polygons}
			(\pgfgoregion{b}{holes} hole).
			Area: \pgfgoregion{b}{area} squares.
		\end{codeexample}
	\end{command}

	\subsection{Moves on Points Already Played}
	\label{subsec:notes}

	In numbered diagrams, when a move is played on a point where there already was a move (for example, when a ko is retaken), books do not draw it: the point still shows the first stone, and below the diagram they write, for example, "3 at 1".

	\begin{key}{/pgfgo/goban/notes=\meta{boolean} (default true, initially false)}

		A numbered move on a point where there already was a numbered move of the same |goban| is not drawn: the point still shows the first one, even if it was captured, and "N at M" is noted. Only numbered moves (|\moves*|) that are visible on the board count. It works with both kernels.

		\begin{codeexample}[]
			\pgfgoset{goban = {5, partial = A2:E4}}

			\begin{goban}[notes]
				\stones[b]{B3,C2,C4}
				\stones[w]{D2,D4,E3}
				\moves*[w]{C3}
				\moves*{D3}
				\moves*[from = 3, w]{C3}
			\end{goban}

			\pgfgomovenotes
		\end{codeexample}
	\end{key}

	\begin{command}{\pgfgomovenotes\opt{\oarg{options}}}
		Writes the notes of the last |goban|, separated by commas. If there are no notes, it writes nothing. The options are those of |/pgfgo/notes|.
	\end{command}

	\begin{key}{/pgfgo/notes = \meta{options}}
		Options of |\pgfgomovenotes| for the whole document, for example |\pgfgoset{notes = {at = en}}|.
	\end{key}

	\begin{key}{/pgfgo/notes/at = \meta{text} (initially at or its translation)}
		Text between the two numbers.
	\end{key}

	\begin{key}{/pgfgo/notes/separator = \meta{text} (initially a comma and a space)}
		Text between the notes.
	\end{key}

	\subsection{Remembering a Board}
	\label{subsec:remember}
	A |goban| with a name stores everything it draws (stones, moves, marks, arrows, lines and territory, also what comes from |\sgf|), together with its options. Another |goban| can start with that board as it was, with |resume|, and |\pgfgoview| shows it in the text (p.~\pageref{subsec:views}).

	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = B2:D4}}

		\begin{goban}[name = board1]
			\stones[b]{C2,D3}
			\moves*[w]{B4}
		\end{goban}

		\begin{goban}[resume = board1]
			\moves*{C3} % Continues with 2 (Black).
		\end{goban}

		\begin{goban}[resume = {board1, numbers}]
			\moves*{C3}
		\end{goban}
	\end{codeexample}

	\begin{key}{/pgfgo/goban/name=\meta{name}}
		Stores with that name everything drawn in this |goban|, together with its options. The name is free (for example, |gob:board1|). Another |goban| with the same name replaces it.
	\end{key}

	\begin{key}{/pgfgo/goban/resume=\meta{name, options}}
		Starts the |goban| with the board that has that name, as it was at the end (with the captures resolved: if that |goban| used |captures|, this one does too). The numbering and the turn continue from there. By default it brings only the stones, without numbers, marks or territory; the options ask for the rest. The board options (size, |partial|, coordinates) are given in each |goban|. If there is no |goban| with that name, an error is raised.

		With |name| and |resume| in the same |goban| (in any order), the new name stores what it brought, as it was drawn, plus its own, and the board of |resume| does not change: this is how variations are made. With the same name in both, the |goban| keeps adding to that board.

		\begin{codeexample}[]
			\pgfgoset{goban = {5, partial = B2:D4}}

			\begin{goban}[name = base]
				\stones[b]{C2,D3}
				\stones[w]{C3}
			\end{goban}

			\begin{goban}[name = variation1, resume = base]
				\moves*[w]{D2}
			\end{goban}

			\begin{goban}[name = variation2, resume = base]
				\moves*[w]{B3}
			\end{goban}

			\begin{goban}[resume = {variation1, numbers}]
				\moves*{B4}
			\end{goban}
		\end{codeexample}
	\end{key}

	\begin{key}{/pgfgo/goban/resume/numbers=\meta{boolean} (default true, initially false)}
		Also brings the move numbers.
	\end{key}

	\begin{key}{/pgfgo/goban/resume/marks=\meta{boolean} (default true, initially false)}
		Also brings the marks, the arrows and the lines.
	\end{key}

	\begin{key}{/pgfgo/goban/resume/territory=\meta{boolean} (default true, initially false)}
		Also brings the territory.
	\end{key}

	So that |resume| does not bring a stone, a move, a mark, an arrow or a territory, it is drawn with the option |forget| (|\stone[forget]{C3}|, |\marks[forget, circle]{A1}|, \ldots); with |label = forget|, a move comes without its number even with |numbers|. What is forgotten is still stored: |\pgfgoview| shows it with |forgotten|.

	\subsection{Views in the Text}
	\label{subsec:views}
	A |goban| with a name stores everything it draws. Later, |\pgfgoview| shows a part of that board, as it was at the end (with the captures done), in the middle of a paragraph: a single stone or a rectangle.

	\begin{codeexample}[]
		\begin{goban}[5, name = board1, captures]
			\stones[b]{B3, C2, C4}
			\stones[w]{D2, D4, E3}
			\moves*[w]{C3, D3}
		\end{goban}

		The black stone at D3 \pgfgoview{board1}{D3} captures the white one at C3 \pgfgoview{board1}{C3}.
		This is the ko: \pgfgoview{board1}{B2:E4}.
	\end{codeexample}

	\begin{command}{\pgfgoview\opt{\oarg{options}}\marg{name}\marg{part}}
		Draws the intersection or the rectangle (written as in |partial|) of the |goban| with that name, as it was at the end. By default everything the |goban| drew in that part is shown (stones, move numbers, marks, arrows and territory) with the background and the coordinates of that |goban|, at the size of the text and centered at the height of the lowercase letters. The options are those of |/pgfgo/view|, which remove the marks, the numbers or the territory, or add what was forgotten, and, also, any |goban| option, which is applied after those of the named |goban|: this removes the background, the coordinates, the grid or the star points.

		A view changes nothing of what the last |goban| stores: |\pgfgoscore|, |\pgfgoprisoners| and |\pgfgomovenotes| still talk about it, and what each named |goban| stores stays the same.

		\begin{codeexample}[]
			\begin{goban}[5, name = board2]
				\stones[b]{B2, C3}
				\moves*[w]{C2, D3}
				\marks[b, triangle]{C3}
			\end{goban}

			\pgfgoview{board2}{B2:D3}
			\pgfgoview[marks = false, numbers = false]{board2}{B2:D3}
			\pgfgoview[label = false, grid = false]{board2}{B2:D3}
			\pgfgoview[background = false]{board2}{B2:D3}
		\end{codeexample}
	\end{command}

	\begin{key}{/pgfgo/view=\meta{options}}
		Options for all views, with |\pgfgoset|. The |goban| options that are not of |/pgfgo/view| are added to the one views come with: |scale = 0.8|.

		\begin{codeexample}[]
			\pgfgoset{view = {background = false, scale = 1}}
			Without background and bigger: \pgfgoview{board2}{C3}.
		\end{codeexample}
	\end{key}

	\begin{key}{/pgfgo/view/marks=\meta{boolean} (default true, initially true)}
		With |false|, the marks, arrows and lines are not drawn.
	\end{key}

	\begin{key}{/pgfgo/view/numbers=\meta{boolean} (default true, initially true)}
		With |false|, the numbers of the moves are not drawn.
	\end{key}

	\begin{key}{/pgfgo/view/territory=\meta{boolean} (default true, initially true)}
		With |false|, the territory is not drawn.
	\end{key}

	\begin{key}{/pgfgo/view/forgotten=\meta{boolean} (default true, initially false)}
		Also draws what was drawn with |forget| (p.~\pageref{subsec:remember}).

		\begin{codeexample}[]
			\begin{goban}[5, name = board3]
				\stones[b]{B2, C3}
				\stone[w, forget]{D3}
			\end{goban}

			\pgfgoview{board3}{B2:D3}
			\pgfgoview[forgotten]{board3}{B2:D3}
		\end{codeexample}
	\end{key}

	\begin{key}{/pgfgo/view/raise=\meta{dimension} (initially 0.5ex)}
		How high above the baseline of the text the center of the view is.
	\end{key}
