\section{Stones, Moves, and Prisoners}

Stones are the fundamental elements. In \pgfgoname, commands and options are provided to draw them flexibly, allowing customization of their appearance and behavior according to the needs of the diagram.

\subsection{Stones}

\begin{command}{\stone\opt{\oarg{options}}\marg{intersection}}
	This command draws an individual stone at the specified intersection. The mandatory argument is the coordinate (e.g., "C2"), while optional options allow adjusting its style, such as color, size, or markings. If no player is given (|b|, |w| or |n|), the stone is \emph{neutral}.
\end{command}

\begin{command}{\stones\opt{\oarg{options}}\marg{intersection \opt{, intersection, ...}}}
	Similar to |\stone|, but allows placing multiple stones using the corresponding syntax (see Coordinate Syntax). If no player is given, the stones are \emph{black}.
\end{command}

Stones, marks and labels that fall outside the visible region of a partial board are not drawn. A coordinate that does not exist on the board (for example, |E9| on a 5$\times$5 board) is ignored and produces a warning.

It is possible to change an option for all players using |\pgfgoset{stone = {...}}|. This is equivalent to |\pgfgoset{players = {all = {stone = {...}}}}|.

\begin{key}{/pgfgo/stone/fill = \meta{options}}
	Controls the fill options of a stone.
\end{key}

\begin{key}{/pgfgo/stone/fill/color = \meta{color}}
	Defines the fill color of a stone.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = {from = B2 to D4}}}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
		\end{goban}
		
		\begin{goban}
			\stones[fill = {color = cyan!80!black}]{C2,D3}
			\stone[fill = {color = magenta!80!black}]{B4}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/fill/opacity = \meta{float} (initially 1)}
	
	Controls the opacity of the stone’s fill, ranging from 0 (transparent) to 1 (opaque).
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = {from = B2 to D4}}}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
		\end{goban}
		
		\begin{goban}
			\stones[b, fill = {opacity = 0.8}]{C2,D3}
			\stone[w, fill = {opacity = 0.5}]{B4}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/line = \meta{option} (initially true)}
	Controls the border options of a stone.
\end{key}

\begin{key}{/pgfgo/stone/line/color = \meta{color} (initially black)}
	
	Sets the border color of the stone, which is black by default for all stones.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = {from = B2 to D4}}}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
		\end{goban}
		
		\begin{goban}
			\stones[b, line = {color = magenta}]{C2,D3}
			\stone[w, line = {color = cyan}]{B4}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/line/opacity = \meta{float} (initially 1)}
	
	Defines the opacity of the stone’s border, allowing it to be more subtle or transparent.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = {from = B2 to D4}}}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
		\end{goban}
		
		\begin{goban}
			\stones[b, line = {opacity = 0.5}]{C2,D3}
			\stone[w, line = {opacity = 0.8}]{B4}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/line/width = \meta{dimension} (initially 0.7pt)}
	
	Controls the thickness of the stone’s border.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = {from = B2 to D4}}}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
		\end{goban}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w, line = {width = 2pt}]{B4}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/radius = \meta{dimension} (initially 5.5pt)}
	
	Sets the size of the stone by its radius.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = {from = B2 to D4}}}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
		\end{goban}
		
		\begin{goban}
			\stones[b, radius = 0.15cm]{C2,D3}
			\stone[w, radius = 0.25cm]{B4}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/scale = \meta{float} (initially 1)}
	
	Defines a scaling factor that affects all elements of the stone (radius, border, markings), maintaining their proportions.
\end{key}

\begin{key}{/pgfgo/stone/black}
	
	Assigns the stone the characteristics of the first player (black by default), including fill and border color. This is the player used by |\stones| unless another one is specified; |\stone|, instead, uses the neutral player.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = {from = B2 to D4}}}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
		\end{goban}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[black]{B4}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/b}
	
	Abbreviation for "black," with identical functionality.
\end{key}

\begin{key}{/pgfgo/stone/white}
	
	Assigns the stone the characteristics of the second player (white by default), with a white fill and black border.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = {from = B2 to D4}}}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[white]{B4}
		\end{goban}
		
		\begin{goban}
			\stones[white]{C2,D3}
			\stone[white]{B4}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/w}
	
	Abbreviation for "white," with the same functionality.
\end{key}

\begin{key}{/pgfgo/stone/neutral}
	The stone belongs to the neutral player. It is the default player of |\stone|.
\end{key}

\begin{key}{/pgfgo/stone/n}
	Abbreviation for "neutral".
\end{key}

\begin{key}{/pgfgo/stone/drop shadow = \meta{options}}
	Shadow of the stone: |true|, |false|, |angle|, |offset| and |opacity|. The options are described in the player reference (p.~\pageref{subsec:players}).

	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = {from = B2 to D4}}}

		\begin{goban}
			\stones[b, drop shadow = {true, angle = -45, offset = 1.5pt}]{C2,D3}
			\stone[w, drop shadow = {true, angle = -45, offset = 1.5pt}]{B4}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/drop shine = \meta{options}}
	Highlight of the stone: |true|, |false|, |angle|, |opacity| and |color|. The options are described in the player reference (p.~\pageref{subsec:players}).

	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = {from = B2 to D4}}}

		\begin{goban}
			\stones[b, drop shine = true]{C2,D3}
			\stone[w, drop shine = {true, color = gray}]{B4}
		\end{goban}
	\end{codeexample}
\end{key}


\begin{key}{/pgfgo/stone/image = \meta{options}}
	Draws the stone with an image (PNG, JPG or PDF) instead of the circle: the fill, the outline and the shine are not drawn, and the image does not need to be round. The shadow (|drop shadow|) has the shape of the image; for that, the image must have transparency (a PNG with an alpha channel, or a PDF): the shadow of a JPG is a rectangle. A prisoner with an image is translucent, like any prisoner.

	Files are looked up like with |\includegraphics| (for example, in the folders of |\graphicspath|). Each file, at each size, is stored only once in the PDF, however many stones use it. The shadow with the shape of the image needs pdf\LaTeX\ or Lua\LaTeX\ (with PDF output); with other engines the stone is drawn without a shadow, with a warning.

	A file name with a comma goes in double braces, |file = {{a,b.png}}|; with single braces, the comma separates two files. As with |\includegraphics|, the name cannot have |&|, |#|, |%|, |$|, |{| or |}|, nor two spaces in a row.

	SVG images cannot be used directly: convert them to PDF first, for example with |inkscape --export-type=pdf stone.svg| or with |rsvg-convert -f pdf -o stone.pdf stone.svg|. The same applies to the background, the grid, the star points and the marks.

	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = {from = B2 to D4}}}

		\begin{goban}
			\stones[b, image = {file = stone-black.png}, drop shadow = {true, angle = -45, offset = 1.5pt}]{C2,D3}
			\stone[w, image = {file = stone-white.png}, drop shadow = {true, angle = -45, offset = 1.5pt}]{B4}
		\end{goban}
	\end{codeexample}

	For all the stones of a player, give the image in the options of the player:

	\begin{codeexample}[]
		\pgfgoset{players = {
			black = {stone = {image = {file = stone-black.png}}},
			white = {stone = {image = {file = stone-white.png}}}}}

		\begin{goban}[5, partial = {from = B2 to D4}]
			\moves{C3, D3, C2, B3}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/image/file = \meta{file \opt{, file, ...}}}
	The image file. With several files, each stone uses one of them, chosen by its position: a stone on a given point always shows the same image, in every diagram. This varies the look of the stones, for example the grain of the white ones.

	\begin{codeexample}[]
		\begin{goban}[5, partial = {from = B2 to D4}]
			\stones[w, image = {file = {stone-white.png, stone-white-2.png}}]{B2, C2, D2, B3, C3, D3}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/image/width = \meta{dimension} (initially the diameter of the stone)}
	The width of the image; the height keeps the proportions. By default it is the diameter of the stone: $2 \times$ |radius| $\times$ |scale|.
\end{key}

\begin{key}{/pgfgo/stone/image/rotate = \meta{angle} or |random| (initially 0)}
	Rotates the image by \meta{angle} degrees, counterclockwise. With |random|, each stone gets a different angle, chosen by its position: a stone on a given point always looks the same, in every diagram and in every frame of an animation. This keeps the grain of a single image from looking repeated. The shadow turns with the image, but still falls to the same side.

	\begin{codeexample}[]
		\begin{goban}[5, partial = {from = B2 to D4}]
			\stones[w, image = {file = stone-white.png,
				rotate = random}]{B2, C2, D2, B3, C3, D3}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{stylekey}{/pgfgo/stone/image/false}
	Removes the image: the stone is a circle again.
\end{stylekey}

\begin{key}{/pgfgo/stone/mark = \meta{mark keys}}
	
	Places a mark on the stone (e.g., triangle, circle). Marks can be further customized with other keys, such as color or size.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = {from = B2 to D4}}}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
		\end{goban}
		
		\begin{goban}
			\stones[b, mark = triangle]{C2,D3}
			\stone[w, mark = t-circle]{B4}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/forget}
	
	When the board uses a memory system (|new remember|, |resume|), this option indicates that the stone should not be remembered, allowing it to be excluded from subsequent states.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = {from = B2 to D4}}}
		
		\begin{goban}[new remember]
			\stones[b]{C2,D3}
			\stone[w, forget]{B4}
		\end{goban}
		
		\begin{goban}[resume]
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/remember}
	
	In a board with active memory, ensures that the stone is remembered even if the general environment uses |forget|.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = {from = B2 to D4}}}
		
		\begin{goban}[forget]
			\stones[b]{C2,D3}
			\stone[w, remember]{B4}
		\end{goban}
		
		\begin{goban}[resume]
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/prisoner=\meta{prisoner options} (default fill opacity = 0.5)}
	
	Characterizes a stone as a prisoner, typically with reduced opacity (0.5 by default) to indicate it has been captured. Other options can be added to customize its appearance.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = {from = B2 to D4}}}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
		\end{goban}
		
		\begin{goban}
			\stones[b, prisoner]{C2,D3}
			\stone[white]{B4}
		\end{goban}
	\end{codeexample}
\end{key}

\subsection{Moves}

Move commands allow representing sequences of plays, automatically alternating between black and white. They include options for numbering the moves and customizing the associated stones.

\begin{command}{\move\opt{*}\opt{\oarg{options}}\marg{intersection}}
	
	Draws a stone as part of a move sequence, automatically alternating colors (black, white, black, etc.). The starred version (|\move*|) adds a numeric label to the stone.
\end{command}

\begin{command}{\moves\opt{*}\opt{\oarg{options}}\marg{intersection \opt{, intersection, ...}}}
	
	Similar to |\move|, but applies the sequence to multiple intersections in a single line.
\end{command}

Within the same |goban|, numbering and turn carry over from one |\moves| to the next. Each new |goban| starts at move 1 with Black; to continue a game in another diagram, use |from| and |b| or |w|, or |resume|.

It is possible to change options for all players using |\pgfgoset{move = {...}}|, which is equivalent to |\pgfgoset{players = {all = {move = {...}}}}|.

\begin{key}{/pgfgo/move/black}
	
	Sets the move sequence to start with the black player, useful for forcing the start of a play in specific diagrams.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = {from = B2 to D4}}}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
		\end{goban}
		
		\begin{goban}
			\moves[b]{C2,B4}
			\move{D3}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/move/white}
	
	Sets the sequence to start with the white player, altering the natural order of moves.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = {from = B2 to D4}}}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
		\end{goban}
		
		\begin{goban}
			\moves[white]{C2,B4}
			\move{D3}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/move/b}
	Short for |black|.
\end{key}

\begin{key}{/pgfgo/move/w}
	Short for |white|.
\end{key}

\begin{key}{/pgfgo/move/from = \meta{number}}
	
	Defines the starting number of a numbered move sequence, allowing continuation from a specific point instead of starting at 1.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = {from = B2 to D4}}}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
		\end{goban}
		
		\begin{goban}
			\moves*[from = 7]{C2,B4}
			\move*{D3}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/move/label = \meta{options}}
	
	Controls the properties of numeric labels in starred moves, such as size, color, or format.
\end{key}

\begin{key}{/pgfgo/move/label/font size=\meta{dimension} (initially automatic)}

	Adjusts the font size of the labels, allowing them to adapt to the stone size or overall design. If it is not set, the size adapts to the number of digits.

	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = {from = B2 to D4}}}

		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}

		\begin{goban}
			\moves*[from = 7]{C2,B4}
			\move*[label = {font size = 0.5em}]{D3}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/move/label/format = \meta{options} (initially arabic)}

\end{key}

\begin{key}{/pgfgo/move/remember}
	
	Ensures that moves are remembered, preserving them for reuse with |resume|.
\end{key}

\subsection{SGF Games}
\label{subsec:sgf}

SGF is the format that almost every Go program uses to save games. With the Lua kernel (p.~\pageref{subsec:kernel}), \pgfgoname\ can read them.

In every command that reads SGF files, the name can have spaces, accents, parentheses and other scripts. The special characters of \TeX\ are written with a backslash: |\sgf{a\&b.sgf}| reads |a&b.sgf|, and the same goes for |\#|, |\%|, |\$|, |\{| and |\}|. Two spaces in a row are read as one, because \TeX\ joins them before \pgfgoname\ sees the name.

\begin{command}{\sgf\opt{\oarg{options}}\marg{file}}

	Draws on the |goban| the main line of the first game in the file: the setup stones (|AB| and |AW|) as with |\stones|, and the moves as with |\moves*|. Variations are not drawn. Passes count for the numbering, but are not drawn. It only exists inside |goban|.
\end{command}

The stones that the nodes after the first move add (|AB|, |AW|) or remove (|AE|) are applied together with the previous move. Removing stones needs |captures|; without |captures|, |\sgf| warns once and does not remove them.

When it ends, the numbering and the turn are ready to continue the game with |\moves|. |\sgf| works with |partial|, |captures|, |remember| and |resume|. If the game is for a board of another size, a warning is produced. With |kernel = expl3|, |\sgf| raises an error.

In the examples, |game.sgf| contains a short 9$\times$9 game:
\begin{verbatim}
(;GM[1]FF[4]SZ[9]PB[Black]PW[White]C[A short game.]
;B[ee];W[ef];B[df];W[ce];B[ff];W[gd];B[eg]C[Black captures.];W[dd];B[cf];W[fd])

\begin{key}{/pgfgo/sgf/from = \meta{number} (initially 1)}

	Number of the first numbered move. The moves before it are drawn without numbers.

	\begin{codeexample}[width=5cm]
		\begin{goban}[captures]
			\sgf[from = 7]{game.sgf}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/sgf/to = \meta{number} (initially the last move)}

	Number of the last move that is played. The following ones are not drawn.

	\begin{codeexample}[width=5cm]
		\begin{goban}[captures]
			\sgf[to = 6]{game.sgf}
			\moves*{F5} % Continues with 7 (Black), somewhere else.
		\end{goban}
	\end{codeexample}
\end{key}


|from| and |to| count the moves of the file. The number that is shown can be different: the |MN| property of the file changes it (the following moves continue from there), and so does |restart|. In the following examples, |variations.sgf| is a 7$\times$7 game with two branches, marks and |MN[10]| on the fourth move of the main line.

\begin{key}{/pgfgo/sgf/variation = \meta{numbers} (initially the main line)}
	Chooses a variation: at each branch of the game, in order, the given variation is taken (1 is the main one). For example, |variation = 2.1| takes the second variation at the first branch and the first one at the second. When the list ends, the first variation is followed. If a variation does not exist, a warning is produced.

	\begin{codeexample}[]
		\begin{goban}[sgf size = variations.sgf]
			\sgf[variation = 2]{variations.sgf}
		\end{goban}

		\begin{goban}[sgf size = variations.sgf]
			\sgf[variation = 1.2]{variations.sgf}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/sgf/marks = \meta{boolean} (default true, initially true)}
	Draws the marks of the file (|TR|, |SQ|, |CR|, |MA|, |SL| and the labels of |LB|), the arrows (|AR|) and the lines (|LN|) of the last node that is shown, and dims the points of |DD| (which stay dimmed until the next |DD|; p.~\pageref{subsec:arrows}). Each mark takes the color of the stone on that point, so that it can be seen, and replaces the number of the move. The territory of the file (|TB| and |TW|) is drawn as that of |\territory| (p.~\pageref{subsec:territory}) and counts for |\pgfgoscore|.

	\begin{codeexample}[]
		\begin{goban}[sgf size = variations.sgf]
			\sgf[to = 2]{variations.sgf}
		\end{goban}

		\begin{goban}[sgf size = variations.sgf]
			\sgf[to = 2, marks = false]{variations.sgf}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/sgf/restart = \meta{number or auto} (initially auto)}
	Restarts the numbering every \meta{number} moves, as books do: with |restart = 100|, move 101 is shown as 1. With 0, it does not restart. With |auto|, it does what the file asks for with |PM|: with |PM[2]|, it restarts every 100 moves; otherwise, it does not restart.

	\begin{codeexample}[]
		\begin{goban}[sgf size = variations.sgf]
			\sgf[]{variations.sgf}
		\end{goban}

		\begin{goban}[sgf size = variations.sgf]
			\sgf[restart = 3]{variations.sgf}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/sgf/numbers = \meta{true, false or auto} (default true, initially auto)}
	With |false|, the moves are drawn without numbers. With |auto|, it does what the file asks for with |PM|: with |PM[0]|, they are not numbered. |\sgfdiagrams| and |\sgfproblem| always number them.

	\begin{codeexample}[]
		\begin{goban}[sgf size = variations.sgf]
			\sgf[numbers = false]{variations.sgf}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/sgf/game = \meta{number} (initially 1)}
	An SGF file can have several games (a collection). It chooses which one is read; if it does not exist, the first one is read with a warning. It also applies to |\sgfanimation|, |\sgfdiagrams|, |\sgfcomment|, |\sgfinfo| and |sgf size|, which have their own |game| \emph{key}; for the whole document, |\pgfgoset{sgf = {game = 2}}|.
\end{key}

\begin{key}{/pgfgo/sgf/variations = \meta{none, children, siblings or auto} (default auto, initially none)}
	Marks with letters the variations of the last move that is shown: with |children|, the moves that can follow; with |siblings|, the alternatives to that move (the letter replaces its number). |auto| does what the file asks for with |ST|. They are only marked if there is more than one.

	\begin{codeexample}[]
		\begin{goban}[sgf size = variations.sgf]
			\sgf[to = 1, marks = false, variations = children]{variations.sgf}
		\end{goban}
	\end{codeexample}
\end{key}

|\sgf| also reads |PL| (who plays next: |\moves| goes on with that color) and |CA| (the charset of the file; see |sgf charset|).

\begin{key}{/pgfgo/sgf charset = \meta{charset} (initially auto)}
	Charset of the SGF files that are not in UTF-8, for every command that reads files. With |auto|, it is the one the file gives with |CA| or, if it gives none, Latin-1 (the default of SGF). |Latin-1|, |GBK| (and |GB2312|), |Big5|, |Shift_JIS|, |EUC-JP| and |EUC-KR| are read; the names ignore case and signs (|shift-jis| and |EUC_KR| work too). A file that is valid UTF-8 is always read as UTF-8, even if |CA| says otherwise.

	Many files from Asian servers do not give their charset; to read them, for example, |\pgfgoset{sgf charset = GBK}|. Another charset produces a warning, and the file is read as Latin-1: it must be converted to UTF-8 first, for example with |iconv -f KOI8-R -t UTF-8 game.sgf > game-utf8.sgf|.

	To write the comments in Chinese, Japanese or Korean, the package for that language is needed too: |ctex|, |luatexja| or |luatexko|. Examples 14, 18 and 19 in |doc/examples| show a commented game in each one.
\end{key}

	Options of every |\sgf| in the document, for example |\pgfgoset{sgf = {variations}}|.
\end{key}

\begin{command}{\sgfanimation\opt{\oarg{options}}\marg{file}}

\begin{key}{/pgfgo/sgf animation/layers = \meta{boolean} (default true, initially true)}

\begin{key}{/pgfgo/sgf problem/crop = \meta{boolean} (default true, initially true)}

\end{key}

\begin{key}{/pgfgo/sgf problem/margin = \meta{number} (initially 1)}
	Lines that |crop| adds around the problem.
\end{key}

\begin{key}{/pgfgo/sgf problem/game = \meta{number} (initially 1)}
	Like |/pgfgo/sgf/game|: in a collection, the problem that is drawn.
\end{key}


\subsection{Layout of the Diagrams}
\label{subsec:layout}

|\sgfdiagrams| and |\sgfproblem| draw several diagrams. By default they go in the paragraph, side by side, and move to the next line when they do not fit. With |columns|, they go in a grid of columns of the same width: each row is a separate line (the page can break between rows), the boards of a row are aligned at the top and an incomplete row leaves its last columns empty. With |fit|, the boards are also scaled to fill the width of their column.

\begin{codeexample}[code only]
	\sgfdiagrams[step = 2, to = 6, layout = {columns = 3, fit}]{game.sgf}
\end{codeexample}

\sgfdiagrams[step = 2, to = 6, layout = {columns = 3, fit}]{game.sgf}

\begin{stylekey}{/pgfgo/layout = \meta{layout options}}
	Layout options for the whole document, for example |\pgfgoset{layout = {columns = 3, fit}}|. For a single command, |/pgfgo/sgf diagrams/layout| and |/pgfgo/sgf problem/layout|; also for the whole document, |\pgfgoset{sgf diagrams = {layout = {...}}}|.
\end{stylekey}

\begin{key}{/pgfgo/layout/columns = \meta{number or auto} (initially auto)}
	Number of columns of the grid. With |auto|, the diagrams go in the paragraph, without a grid.
\end{key}

\begin{key}{/pgfgo/layout/width = \meta{dimension} (initially \texttt{\string\linewidth})}
	Width of the grid. If it is less than the width of the line, the grid is centered.
\end{key}

\begin{key}{/pgfgo/layout/column sep = \meta{dimension} (initially 1em)}
	Space between columns.
\end{key}

\begin{key}{/pgfgo/layout/row sep = \meta{skip} (initially \texttt{\string\medskipamount})}
	Space between rows.
\end{key}

\begin{key}{/pgfgo/layout/fit = \meta{boolean} (default true, initially false)}
	Scales each board to fill the width of its column; it replaces the |scale| of the board. The lines keep their width. In |\sgfdiagrams| and |\sgfproblem|, all the boards use the scale that fills the column with the first one, so the stones have the same size in all of them. Only with |columns|.
\end{key}

\begin{key}{/pgfgo/layout/align = \meta{center, left or right} (initially center)}
	Where each diagram goes within its column, with its caption and its notes. The comments are always aligned to the left.
\end{key}

\begin{environment}{{pgfgolayout}\opt{\oarg{layout options}}}
	Lays out in a grid the |goban|s it contains, and the diagrams of the |\sgfdiagrams| and |\sgfproblem| it contains, in the order in which they appear. So boards drawn by hand and diagrams of different files can be put together in a grid. The options are those of |/pgfgo/layout|; the |layout| options of the commands inside are not used. There must be no text inside, only the boards and the commands. With |fit|, each |goban| is drawn several times to measure it: its contents should have no effects outside the board (for example, counters that are stepped).

	\begin{codeexample}[code only]
		\begin{pgfgolayout}[columns = 4, fit]
			\begin{goban}[5]\stones{C3}\end{goban}
			\begin{goban}[5]\stones[w]{C3}\end{goban}
			\sgfdiagrams[step = 5]{game.sgf}
		\end{pgfgolayout}
	\end{codeexample}
\end{environment}

\begin{pgfgolayout}[columns = 4, fit]
	\begin{goban}[5]\stones{C3}\end{goban}
	\begin{goban}[5]\stones[w]{C3}\end{goban}
	\sgfdiagrams[step = 5]{game.sgf}
\end{pgfgolayout}



\subsection{Prisoners}

\pgfgoname\ provides specific commands to represent prisoners, typically with reduced opacity to distinguish them from active stones.

\begin{command}{\prisoner\opt{\oarg{options}}\marg{intersection}}
	
	Draws a prisoner stone at a specific intersection, with options to customize its style.
\end{command}

\begin{command}{\prisoners\opt{\oarg{options}}\marg{intersection \opt{, intersection, ...}}}
	
	Places multiple prisoner stones in a single line, efficient for showing captured groups.
\end{command}

\begin{key}{/pgfgo/prisoner=\meta{stone options} (default fill opacity = 0.5)}
	
	Defines the characteristics of prisoner stones, with a default fill opacity of 0.5 to indicate their status. Other |\stone| options, such as color or size, can be applied.
	
\end{key}
