\section{Stones, Moves, and Prisoners}

Stones are the fundamental elements. In \pgfgoname, commands and options are provided to draw them flexibly, allowing customization of their appearance and behavior according to the needs of the diagram.

\subsection{Stones}

\begin{command}{\stone\opt{\oarg{options}}\marg{intersection}}
	This command draws an individual stone at the specified intersection. The mandatory argument is the coordinate (e.g., "C2"), while optional options allow adjusting its style, such as color, size, or markings. If no player is given (|b|, |w| or |n|), the stone is \emph{neutral}.
\end{command}

\begin{command}{\stones\opt{\oarg{options}}\marg{intersection \opt{, intersection, ...}}}
	Similar to |\stone|, but allows placing multiple stones using the corresponding syntax (see Coordinate Syntax). If no player is given, the stones are also \emph{neutral}: black stones are written |\stones[b]{...}|.
\end{command}

Stones, marks and labels that fall outside the visible region of a partial board are not drawn. A coordinate that does not exist on the board (for example, |E9| on a 5$\times$5 board) is ignored and produces a warning.

It is possible to change an option for all players using |\pgfgoset{stone = {...}}|. This is equivalent to |\pgfgoset{players = {all = {stone = {...}}}}|.

\begin{key}{/pgfgo/stone/fill = \meta{options}}
	Controls the fill options of a stone.
\end{key}

\begin{key}{/pgfgo/stone/fill/color = \meta{color}}
	Defines the fill color of a stone.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = B2:D4}}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
		\end{goban}
		
		\begin{goban}
			\stones[fill = {color = cyan!80!black}]{C2,D3}
			\stone[fill = {color = magenta!80!black}]{B4}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/fill/opacity = \meta{float} (initially 1)}
	
	Controls the opacity of the stone’s fill, ranging from 0 (transparent) to 1 (opaque).
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = B2:D4}}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
		\end{goban}
		
		\begin{goban}
			\stones[b, fill = {opacity = 0.8}]{C2,D3}
			\stone[w, fill = {opacity = 0.5}]{B4}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/line = \meta{option} (initially true)}
	Controls the border options of a stone.
\end{key}

\begin{key}{/pgfgo/stone/line/color = \meta{color} (initially black)}
	
	Sets the border color of the stone, which is black by default for all stones.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = B2:D4}}

			\stones[b]{C2,D3}
			\stone[w]{B4}
		\end{goban}
		
		\begin{goban}
			\stones[b, line = {color = magenta}]{C2,D3}
			\stone[w, line = {color = cyan}]{B4}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/line/opacity = \meta{float} (initially 1)}
	
	Defines the opacity of the stone’s border, allowing it to be more subtle or transparent.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = B2:D4}}

			\stones[b]{C2,D3}

		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/line/width = \meta{dimension} (initially 0.7pt)}

\begin{key}{/pgfgo/stone/radius = \meta{dimension} (initially 5.5pt)}
	
	Sets the size of the stone by its radius.
	
	\begin{codeexample}[]

		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
		\end{goban}

		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/scale = \meta{float} (initially 1)}
	
	Defines a scaling factor that affects all elements of the stone (radius, border, markings), maintaining their proportions.

	\begin{codeexample}[]
		\begin{goban}[5, partial = B2:D4]
			\stones[b, mark = circle]{C2}
			\stones[b, scale = 0.7, mark = circle]{D3}
			\stone[w, scale = 1.3, mark = triangle]{B4}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/black}
	
	Assigns the stone the characteristics of the first player (black by default), including fill and border color. Without |black| (or |b|), |\stone| and |\stones| draw neutral stones.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = B2:D4}}

		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/b}
	
	Abbreviation for "black," with identical functionality.
\end{key}

\begin{key}{/pgfgo/stone/white}
	
	Assigns the stone the characteristics of the second player (white by default), with a white fill and black border.
	
	\begin{codeexample}[]

		\begin{goban}
			\stones[b]{C2,D3}

		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/w}
	
	Abbreviation for "white," with the same functionality.
\end{key}

\begin{key}{/pgfgo/stone/neutral}
	The stone belongs to the neutral player. It is the default player of |\stone| and |\stones|.

	\begin{codeexample}[]
		\begin{goban}[5, partial = B2:D4]
			\stones[b]{C2}
			\stones[w]{D3}
			\stone{B4}
			\stone[n]{C3}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/n}
	Abbreviation for "neutral".
\end{key}

\begin{key}{/pgfgo/stone/drop shadow = \meta{options}}
	Shadow of the stone: |true|, |false|, |angle|, |offset|, |opacity| and |color|. The options are described in the player reference (p.~\pageref{subsec:players}).

	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = B2:D4}}

		\begin{goban}
			\stones[b, drop shadow = {true, angle = -45, offset = 1.5pt}]{C2,D3}
			\stone[w, drop shadow = {true, angle = -45, offset = 1.5pt}]{B4}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/shine = \meta{options} (default true)}
	Highlight of the stone, drawn on top of it: |true|, |false|, |angle|, |opacity|, |color| and |image|. The options are described in the player reference (p.~\pageref{subsec:players}).

	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = B2:D4}}

		\begin{goban}
			\stones[b, shine]{C2,D3}
			\stone[w, shine = {true, color = gray}]{B4}
		\end{goban}
	\end{codeexample}

	With |image|, the shine is an image with transparency (PNG or PDF), centered on the stone. If the stone is an image too, the shine is drawn on top of it and does not turn with it: the stones can have their grain turned at random (|rotate = random|) and the light always on the same side.

	\begin{codeexample}[]
		\begin{goban}[5, partial = B2:D4]
			\stones[w, image = {file = stone-white.png, rotate = random},
				shine = {true, image = {file = shine.png}}]{B2, C2, D2, B3, C3, D3}
		\end{goban}
	\end{codeexample}
\end{key}


\label{subsec:stoneimages}\begin{key}{/pgfgo/stone/image = \meta{options}}
	Draws the stone with an image (PNG, JPG or PDF) instead of the circle: the fill and the outline are not drawn, and the image does not need to be round. If the shine (|shine|) is on, it is drawn on top of the image. The shadow (|drop shadow|) has the shape of the image; for that, the image must have transparency (a PNG with an alpha channel, or a PDF): the shadow of a JPG is a rectangle. A prisoner with an image is translucent, like any prisoner.

	Files are looked up like with |\includegraphics| (for example, in the folders of |\graphicspath|). Each file, at each size, is stored only once in the PDF, however many stones use it. The shadow with the shape of the image needs pdf\LaTeX\ or Lua\LaTeX\ (with PDF output); with other engines the stone is drawn without a shadow, with a warning.

	A file name with a comma goes in double braces, |file = {{a,b.png}}|; with single braces, the comma separates two files. As with |\includegraphics|, the name cannot have |&|, |#|, |%|, |$|, |{| or |}|, nor two spaces in a row.

	SVG images cannot be used directly: convert them to PDF first, for example with |inkscape --export-type=pdf stone.svg| or with |rsvg-convert -f pdf -o stone.pdf stone.svg|. The same applies to the background, the grid, the star points and the marks.

	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = B2:D4}}

		\begin{goban}
			\stones[b, image = {file = stone-black.png}, drop shadow = {true, angle = -45, offset = 1.5pt}]{C2,D3}
			\stone[w, image = {file = stone-white.png}, drop shadow = {true, angle = -45, offset = 1.5pt}]{B4}
		\end{goban}
	\end{codeexample}

	For all the stones of a player, give the image in the options of the player:

	\begin{codeexample}[]
		\pgfgoset{players = {
			black = {stone = {image = {file = stone-black.png}}},
			white = {stone = {image = {file = stone-white.png}}}}}

		\begin{goban}[5, partial = B2:D4]
			\moves{C3, D3, C2, B3}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/image/file = \meta{file \opt{, file, ...}}}
	The image file. With several files, each stone uses one of them, chosen by its position: a stone on a given point always shows the same image, in every diagram. This varies the look of the stones, for example the grain of the white ones.

	\begin{codeexample}[]
		\begin{goban}[5, partial = B2:D4]
			\stones[w, image = {file = {stone-white.png, stone-white-2.png}}]{B2, C2, D2, B3, C3, D3}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/image/width = \meta{dimension} (initially the diameter of the stone)}
	The width of the image; the height keeps the proportions. By default it is the diameter of the stone: $2 \times$ |radius| $\times$ |scale|.
\end{key}

\begin{key}{/pgfgo/stone/image/rotate = \meta{angle}, |random| or \meta{min}|:|\meta{max} (initially 0)}
	Rotates the image by \meta{angle} degrees, counterclockwise. With |random|, each stone gets a different angle, chosen by its position: a stone on a given point always looks the same, in every diagram and in every frame of an animation. This keeps the grain of a single image from looking repeated. The shadow turns with the image, but still falls to the same side.

	\begin{codeexample}[]
		\begin{goban}[5, partial = B2:D4]
			\stones[w, image = {file = stone-white.png,
				rotate = random}]{B2, C2, D2, B3, C3, D3}
		\end{goban}
	\end{codeexample}

	With a range \meta{min}|:|\meta{max}, the angle is chosen as with |random|, but between \meta{min} and \meta{max} degrees (|random| is the same as |0:360|). This is useful for images that cannot be turned all the way, such as a stone lit from one side.

	\begin{codeexample}[]
		\begin{goban}[5, partial = B2:D4]
			\stones[w, image = {file = stone-white.png,
				rotate = -20:20}]{B2, C2, D2, B3, C3, D3}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/image/opacity = \meta{float} (initially 1)}
	The opacity of the image.

	\begin{codeexample}[]
		\begin{goban}[5, partial = B2:D4]
			\stones[b, image = {file = stone-black.png}]{C2}
			\stones[b, image = {file = stone-black.png, width = 8pt}]{D3}
			\stone[w, image = {file = stone-white.png, opacity = 0.5}]{B4}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{stylekey}{/pgfgo/stone/image/false}
	Removes the image: the stone is a circle again.
\end{stylekey}

\begin{key}{/pgfgo/stone/mark = \meta{mark keys}}
	
	Places a mark on the stone (e.g., triangle, circle). Marks can be further customized with other keys, such as color or size.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = B2:D4}}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
		\end{goban}
		
		\begin{goban}
			\stones[b, mark = triangle]{C2,D3}
			\stone[w, mark = t-circle]{B4}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/forget}

	|resume| does not bring this stone (p.~\pageref{subsec:remember}). It stays in the memory of the named |goban|: |\pgfgoview| shows it with |forgotten|.

	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = B2:D4}}

		\begin{goban}[name = a]
			\stones[b]{C2,D3}
			\stone[w, forget]{B4}
		\end{goban}

		\begin{goban}[resume = a]
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/stone/prisoner=\meta{prisoner options} (default fill opacity = 0.5)}
	
	Characterizes a stone as a prisoner, typically with reduced opacity (0.5 by default) to indicate it has been captured. Other options can be added to customize its appearance: they are applied after the prisoner style (|/pgfgo/prisoner| and |/pgfgo/players/white/stone/prisoner|).
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = B2:D4}}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
		\end{goban}
		
		\begin{goban}
			\stones[b, prisoner]{C2,D3}
			\stone[white]{B4}
		\end{goban}
	\end{codeexample}
\end{key}

\subsection{Moves}

Move commands allow representing sequences of plays, automatically alternating between black and white. They include options for numbering the moves and customizing the associated stones.

\begin{command}{\move\opt{*}\opt{\oarg{options}}\marg{intersection}}
	
	Draws a stone as part of a move sequence, automatically alternating colors (black, white, black, etc.). The starred version (|\move*|) adds a numeric label to the stone.
\end{command}

\begin{command}{\moves\opt{*}\opt{\oarg{options}}\marg{intersection \opt{, intersection, ...}}}
	
	Similar to |\move|, but with several moves. The moves are a sequence: they are played in the order in which they are written, and an intersection can be repeated (for example, to retake a ko with |captures|). Each item of the list is an intersection (|C3|), a path (|A1 -- A3 -- C3|: each stretch goes from its first point to the second) or a part of a row or a column, in the order in which it is written (|C:A 4| is |C4 -- A4|; |C 5:1| is |C5 -- C1|). The other forms of |\stones| (rectangles, |*|, brackets, |;|) have no clear order and raise an error.

	\begin{codeexample}[]
		\begin{goban}[5]
			\moves*{B1 -| D3, C:A 4, E5}
		\end{goban}
	\end{codeexample}
\end{command}

Within the same |goban|, numbering and turn carry over from one |\moves| to the next. Each new |goban| starts at move 1 with Black; to continue a game in another diagram, use |from| and |b| or |w|, or |resume|.

The options of |\moves| also accept those of the style of the stone (|fill|, |line|, |radius|, |shine|, |image|, etc.) and |mark|, like |\stones|. For example, to mark the last move:

\begin{codeexample}[]
	\begin{goban}[5, partial = B2:D4]
		\moves{C3, D3, C2}
		\moves[mark = circle]{B3}
	\end{goban}
\end{codeexample}

It is possible to change options for all players using |\pgfgoset{move = {...}}|, which is equivalent to |\pgfgoset{players = {all = {move = {...}}}}|.

\begin{key}{/pgfgo/move/black}
	
	Sets the move sequence to start with the black player, useful for forcing the start of a play in specific diagrams.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = B2:D4}}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
		\end{goban}
		
		\begin{goban}
			\moves[b]{C2,B4}
			\move{D3}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/move/white}
	
	Sets the sequence to start with the white player, altering the natural order of moves.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = B2:D4}}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
		\end{goban}
		
		\begin{goban}
			\moves[white]{C2,B4}
			\move{D3}
		\end{goban}
	\end{codeexample}
\end{key}

\end{key}

			\move*{D3}
		\end{goban}
		
		\begin{goban}
			\moves*[label = {color = red}]{C2,B4}
			\move*[label = {color = blue}]{D3}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/move/label/forget}

	With |resume = {..., numbers}|, this move comes without its number (p.~\pageref{subsec:remember}). With |\pgfgoview|, the number is only shown with |forgotten|.

	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = B2:D4}}

		\begin{goban}[name = a]
			\moves*{C2}
			\moves*[label = forget]{B4}
			\moves*{D3}
		\end{goban}

		\begin{goban}[resume = {a, numbers}]
			\moves*{C:D4}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/move/forget}

	|resume| does not bring these moves, neither the stone nor the number. The numbering continues from the last move it brings.
\end{key}

\subsection{SGF Games}
\label{subsec:sgf}

SGF is the format that almost every Go program uses to save games. With the Lua kernel (p.~\pageref{subsec:kernel}), \pgfgoname\ can read them.

In every command that reads SGF files, the name can have spaces, accents, parentheses and other scripts. The special characters of \TeX\ are written with a backslash: |\sgf{a\&b.sgf}| reads |a&b.sgf|, and the same goes for |\#|, |\%|, |\$|, |\{| and |\}|. Two spaces in a row are read as one, because \TeX\ joins them before \pgfgoname\ sees the name.

\begin{command}{\sgf\opt{\oarg{options}}\marg{file}}

	Draws on the |goban| the main line of the first game in the file: the setup stones (|AB| and |AW|) as with |\stones|, and the moves as with |\moves*|. Variations are not drawn. Passes count for the numbering, but are not drawn. It only exists inside |goban|.
\end{command}

The stones that the nodes after the first move add (|AB|, |AW|) or remove (|AE|) are applied together with the previous move. Removing stones needs |captures|; without |captures|, |\sgf| warns once and does not remove them.

When it ends, the numbering and the turn are ready to continue the game with |\moves|. |\sgf| works with |partial|, |captures|, |name| and |resume|. If the game is for a board of another size, a warning is produced. With |kernel = expl3|, |\sgf| raises an error.

In the examples, |game.sgf| contains a short 9$\times$9 game:
\begin{verbatim}
(;GM[1]FF[4]SZ[9]PB[Black]PW[White]C[A short game.]
;B[ee];W[ef];B[df];W[ce];B[ff];W[gd];B[eg]C[Black captures.];W[dd];B[cf];W[fd])
\end{verbatim}

\begin{codeexample}[width=5cm]
	\begin{goban}[label = {at = {north, west}}]
		\sgf{game.sgf} % Without captures, 2 stays.
	\end{goban}
\end{codeexample}

\begin{codeexample}[width=5cm]
	\begin{goban}[label = {at = {north, west}}, captures]
		\sgf{game.sgf}
	\end{goban}
\end{codeexample}

\begin{key}{/pgfgo/sgf/from = \meta{number} (initially 1)}

	Number of the first numbered move. The moves before it are drawn without numbers.

	\begin{codeexample}[width=5cm]
		\begin{goban}[captures]
			\sgf[from = 7]{game.sgf}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/sgf/to = \meta{number} or |last| (initially last)}

	Number of the last move that is played. The following ones are not drawn. With |last| (or a negative number), up to the last one.

	\begin{codeexample}[width=5cm]
		\begin{goban}[captures]
			\sgf[to = 6]{game.sgf}
			\moves*{F5} % Continues with 7 (Black), somewhere else.
		\end{goban}
	\end{codeexample}
\end{key}


|from| and |to| count the moves of the file. The number that is shown can be different: the |MN| property of the file changes it (the following moves continue from there), and so does |restart|. In the following examples, |variations.sgf| is a 7$\times$7 game with two branches, marks and |MN[10]| on the fourth move of the main line.

\begin{key}{/pgfgo/sgf/variation = \meta{numbers} (initially the main line)}
	Chooses a variation: at each branch of the game, in order, the given variation is taken (1 is the main one). For example, |variation = 2.1| takes the second variation at the first branch and the first one at the second. When the list ends, the first variation is followed. If a variation does not exist, a warning is produced.

	\begin{codeexample}[]
		\begin{goban}[sgf size = variations.sgf]
			\sgf[variation = 2]{variations.sgf}
		\end{goban}

		\begin{goban}[sgf size = variations.sgf]
			\sgf[variation = 1.2]{variations.sgf}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/sgf/marks = \meta{boolean} (default true, initially true)}
	Draws the marks of the file (|TR|, |SQ|, |CR|, |MA|, |SL| and the labels of |LB|), the arrows (|AR|) and the lines (|LN|) of the last node that is shown, and dims the points of |DD| (which stay dimmed until the next |DD|; p.~\pageref{subsec:arrows}). Each mark takes the color of the stone on that point, so that it can be seen, and replaces the number of the move. The territory of the file (|TB| and |TW|) is drawn as that of |\territory| (p.~\pageref{subsec:territory}) and counts for |\pgfgoscore|.

	\begin{codeexample}[]
		\begin{goban}[sgf size = variations.sgf]
			\sgf[to = 2]{variations.sgf}
		\end{goban}

		\begin{goban}[sgf size = variations.sgf]
			\sgf[to = 2, marks = false]{variations.sgf}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/sgf/step = \meta{number} or \meta{list of numbers} (initially 100)}
	The size of the stretches in which |restart| restarts the numbering. With a list, each number is the size of a stretch, and the last one is repeated: with |step = {50, 100}|, the first stretch has 50 moves and the others 100. It is the same \emph{key} as |/pgfgo/sgf diagrams/step|, where each stretch is a diagram.
\end{key}

\begin{key}{/pgfgo/sgf/restart = \meta{restart options} (default true)}
	From the second stretch of |step| moves on, each one is numbered from 1 (or from |restart/from|), as books do: with |step = 100, restart|, move 101 is shown as 1. The first stretch keeps the numbers of the game. |\sgfanimation| and |\sgfdiagrams| have the same options.

	\begin{codeexample}[]
		\begin{goban}[sgf size = variations.sgf]
			\sgf[]{variations.sgf}
		\end{goban}

		\begin{goban}[sgf size = variations.sgf]
			\sgf[step = 3, restart]{variations.sgf}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/sgf/restart/true}
	Restarts the numbering in each stretch.
\end{key}

\begin{key}{/pgfgo/sgf/restart/false}
	Does not restart it: the moves have the numbers of the game (with |MN|).
\end{key}

\begin{key}{/pgfgo/sgf/restart/auto}
	Does what the file asks for with |PM|: with |PM[2]|, the numbering restarts every 100 moves, whatever |step| is (move 101 is shown as 1); otherwise, it does not restart. It is the initial value.
\end{key}

\begin{key}{/pgfgo/sgf/restart/from = \meta{number} (initially 1)}
	The number of the first move of each stretch, from the second one on. It also turns |restart| on. With |step = 99| and |restart = {from = 0}|, move 100 is shown as 0, 101 as 1, and so on.

	\begin{codeexample}[]
		\begin{goban}[sgf size = variations.sgf]
			\sgf[step = 3, restart = {from = 0}]{variations.sgf}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/sgf/numbers = \meta{true, false or auto} (default true, initially auto)}
	With |false|, the moves are drawn without numbers. With |auto|, it does what the file asks for with |PM|: with |PM[0]|, they are not numbered. |\sgfdiagrams| has its own |numbers| \emph{key}, and |\sgfproblem| always numbers them.

	\begin{codeexample}[]
		\begin{goban}[sgf size = variations.sgf]
			\sgf[numbers = false]{variations.sgf}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/sgf/last mark = \meta{mark options} or |false| (initially false)}
	Marks the last move shown, with the options of a mark (|circle|, |triangle|, |custom = ...|, etc.). A pass is not marked, and neither is a point the file already marks. The mark goes over the number: it is mostly useful with |numbers = false|.

	\begin{codeexample}[]
		\begin{goban}[sgf size = variations.sgf]
			\sgf[numbers = false, last mark = circle]{variations.sgf}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/sgf/game = \meta{number} (initially 1)}
	An SGF file can have several games (a collection). It chooses which one is read; if it does not exist, the first one is read with a warning. It also applies to |\sgfanimation|, |\sgfdiagrams|, |\sgfcomment|, |\sgfinfo| and |sgf size|, which have their own |game| \emph{key}; for the whole document, |\pgfgoset{sgf = {game = 2}}|.
\end{key}

\begin{key}{/pgfgo/sgf/variations = \meta{none, children, siblings or auto} (default auto, initially none)}
	Marks with letters the variations of the last move that is shown: with |children|, the moves that can follow; with |siblings|, the alternatives to that move (the letter replaces its number). |auto| does what the file asks for with |ST|. They are only marked if there is more than one.

	\begin{codeexample}[]
		\begin{goban}[sgf size = variations.sgf]
			\sgf[to = 1, marks = false, variations = children]{variations.sgf}
		\end{goban}
	\end{codeexample}
\end{key}

|\sgf| also reads |PL| (who plays next: |\moves| goes on with that color) and |CA| (the charset of the file; see |sgf charset|).

\begin{key}{/pgfgo/sgf charset = \meta{charset} (initially auto)}
	Charset of the SGF files that are not in UTF-8, for every command that reads files. With |auto|, it is the one the file gives with |CA| or, if it gives none, Latin-1 (the default of SGF). |Latin-1|, |GBK| (and |GB2312|), |Big5|, |Shift_JIS|, |EUC-JP| and |EUC-KR| are read; the names ignore case and signs (|shift-jis| and |EUC_KR| work too). A file that is valid UTF-8 is always read as UTF-8, even if |CA| says otherwise.

	Many files from Asian servers do not give their charset; to read them, for example, |\pgfgoset{sgf charset = GBK}|. Another charset produces a warning, and the file is read as Latin-1: it must be converted to UTF-8 first, for example with |iconv -f KOI8-R -t UTF-8 game.sgf > game-utf8.sgf|.

	To write the comments in Chinese, Japanese or Korean, the package for that language is needed too: |ctex|, |luatexja| or |luatexko|. Examples 14, 18 and 19 in |doc/examples| show a commented game in each one.
\end{key}

\begin{key}{/pgfgo/sgf = \meta{sgf options}}
	Options of every |\sgf| in the document, for example |\pgfgoset{sgf = {variations}}|.
\end{key}

\begin{command}{\sgfinfo\opt{\oarg{options}}\marg{file}\marg{property}}
	Writes a property of the root of the game, for example |PB| and |PW| (the players), |RE| (the result), |KM| (the komi) or |DT| (the date). If it is missing, it writes nothing. It can be used outside |goban|.

	\begin{codeexample}[]
		\sgfinfo{variations.sgf}{PB} (Black) against \sgfinfo{variations.sgf}{PW} (White), komi \sgfinfo{variations.sgf}{KM}. Result: \sgfinfo{variations.sgf}{RE}.
	\end{codeexample}
\end{command}

\begin{command}{\sgfcomment\opt{\oarg{options}}\marg{file}\marg{move}}
	Writes the comment of a move of the file (|0|: the one before the first move): the name of the node (|N|), in bold; the annotations of the move (|TE|, |BM|, |DO|, |IT|) and of the position (|GB|, |GW|, |DM|, |UC|); and the comment (|C|). The annotations are written with the texts of |/pgfgo/sgf annotations|. Values (|V|) and hotspots (|HO|), which are meant for the game tree, are not written. It can be used outside |goban|. |\sgfanimation| and |\sgfdiagrams| write the comments in the same way.

	\begin{codeexample}[]
		Move 7: \sgfcomment{game.sgf}{7}
	\end{codeexample}
\end{command}

\begin{key}{/pgfgo/sgf info/game = \meta{number} (initially 1)}
	Like |/pgfgo/sgf/game|, for |\sgfinfo|.
\end{key}

\begin{key}{/pgfgo/sgf comment/game = \meta{number} (initially 1)}
	Like |/pgfgo/sgf/game|.
\end{key}

\begin{key}{/pgfgo/sgf comment/variation = \meta{numbers} (initially the main line)}
	Like |/pgfgo/sgf/variation|.

	\begin{codeexample}[]
		\sgfcomment[variation = 2]{problem.sgf}{1}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/sgf comment/annotations = \meta{boolean} (default true, initially true)}
	Writes the name and the annotations. With |false|, only the comment.

	\begin{codeexample}[]
		With: \sgfcomment{game.sgf}{7}
		\quad Without: \sgfcomment[annotations = false]{game.sgf}{7}
	\end{codeexample}
\end{key}

The texts of the annotations are in the language of the document (p.~\pageref{subsec:languages}), and they can be changed for the whole document with |\pgfgoset{sgf annotations = {...}}|, for example |bad move = Mala jugada|. The emphasized annotations of |DM| and |UC| use the same text.

\begin{key}{/pgfgo/sgf annotations/tesuji = \meta{text} (initially Tesuji or its translation)}
	Text of |TE|.

	\begin{codeexample}[]
		\pgfgoset{sgf annotations = {tesuji = Good move}}
		\sgfcomment{game.sgf}{7}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/sgf annotations/very good tesuji = \meta{text} (initially Very good tesuji or its translation)}
	Text of |TE[2]|.
\end{key}

\begin{key}{/pgfgo/sgf annotations/bad move = \meta{text} (initially Bad move or its translation)}
	Text of |BM|.
\end{key}

\begin{key}{/pgfgo/sgf annotations/very bad move = \meta{text} (initially Very bad move or its translation)}
	Text of |BM[2]|.
\end{key}

\begin{key}{/pgfgo/sgf annotations/doubtful = \meta{text} (initially Doubtful move or its translation)}
	Text of |DO|.
\end{key}

\begin{key}{/pgfgo/sgf annotations/interesting = \meta{text} (initially Interesting move or its translation)}
	Text of |IT|.
\end{key}

\begin{key}{/pgfgo/sgf annotations/good for black = \meta{text} (initially Good for Black or its translation)}
	Text of |GB|.
\end{key}

\begin{key}{/pgfgo/sgf annotations/very good for black = \meta{text} (initially Very good for Black or its translation)}
	Text of |GB[2]|.
\end{key}

\begin{key}{/pgfgo/sgf annotations/good for white = \meta{text} (initially Good for White or its translation)}
	Text of |GW|.
\end{key}

\begin{key}{/pgfgo/sgf annotations/very good for white = \meta{text} (initially Very good for White or its translation)}
	Text of |GW[2]|.
\end{key}

\begin{key}{/pgfgo/sgf annotations/even = \meta{text} (initially Even position or its translation)}
	Text of |DM|.
\end{key}

\begin{key}{/pgfgo/sgf annotations/unclear = \meta{text} (initially Unclear position or its translation)}
	Text of |UC|.
\end{key}

\begin{command}{\sgfanimation\opt{\oarg{options}}\marg{file}}

	Steps through the game with the |animate| package: it draws one |goban| per move, and the |animate| buttons go forward, back or play it. Frame $k$ shows the position after move $k$; frame 0, the position before the first move. It is used outside |goban|. The boards have the size of the game (|SZ|), unless |goban| gives another one. It needs |\usepackage{animate}| and two runs.
\end{command}

|animate| animations work in Acrobat Reader and in a few other viewers (Foxit, PDF-XChange). Other viewers only show the first frame. These options can be set for the whole document with |\pgfgoset{sgf animation = {...}}|.

\begin{codeexample}[width=5cm]
	\sgfanimation[comments, goban = {captures}]{game.sgf}
\end{codeexample}

\begin{key}{/pgfgo/sgf animation/from = \meta{number} (initially 0)}
	First frame.
\end{key}

\begin{key}{/pgfgo/sgf animation/to = \meta{number} or |last| (initially last)}
	Last frame. With |last| (or |-1|), the frame of the last move.
\end{key}

\begin{key}{/pgfgo/sgf animation/fps = \meta{number} (initially 1)}
	Frames per second when playing.
\end{key}

\begin{key}{/pgfgo/sgf animation/goban = \meta{goban options}}
	Options of every board, as in |\begin{goban}[...]|.
\end{key}

\begin{key}{/pgfgo/sgf animation/animate = \meta{animate options} (initially controls)}
	Options of the |animateinline| environment of |animate|, for example |controls, loop|.
\end{key}

\begin{key}{/pgfgo/sgf animation/numbers = \meta{boolean} (default true, initially false)}
	Numbers every move, as |\sgf| does. Without |numbers|, moves have no numbers and the last one is marked.

	\begin{codeexample}[width=5cm]
		\sgfanimation[from = 6, to = 8, numbers, goban = {captures}]{game.sgf}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/sgf animation/last mark = \meta{mark options or false} (initially circle)}
	Mark of the last move, in the colour of the player who made it. With |false|, the last move is not marked. It is not used with |numbers|, nor when the file already marks that point.

	\begin{codeexample}[width=5cm]
		\sgfanimation[from = 5, to = 7, last mark = triangle, fps = 2,
			animate = {controls, loop}, goban = {captures}]{game.sgf}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/sgf animation/comments = \meta{comments options} (default true)}
	Shows the comment (|C|) of the move below the board. The one of frame 0 is the comment of the nodes before the first move. It is turned on with |comments| or |comments = true| and off with |false| (the initial value); |height| and |font| do not turn it on: |comments = {true, height = 6em}|.
\end{key}

\begin{stylekey}{/pgfgo/sgf animation/comments/true}
	Shows the comments.
\end{stylekey}

\begin{stylekey}{/pgfgo/sgf animation/comments/false}
	Does not show the comments. It is the initial value.
\end{stylekey}

\begin{key}{/pgfgo/sgf animation/comments/height = \meta{dimension} (initially 4em)}
	Height of the space for the comments. It is the same in every frame, so that frames do not change size.
\end{key}

\begin{key}{/pgfgo/sgf animation/comments/font = \meta{font commands} (initially \texttt{\string\small})}
	Font of the comments.

	\begin{codeexample}[width=5cm]
		\sgfanimation[from = 7, to = 9, goban = {captures},
			comments = {true, height = 2.5em, font = \itshape}]{game.sgf}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/sgf animation/variation = \meta{numbers} (initially the main line)}
	Like |/pgfgo/sgf/variation|: the animation follows the line that leads to that variation.

	\begin{codeexample}[width=5cm]
		\sgfanimation[variation = 2, numbers]{variations.sgf}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/sgf animation/marks = \meta{boolean} (default true, initially true)}
	Like |/pgfgo/sgf/marks|: each frame shows the marks of its move.
\end{key}

\begin{key}{/pgfgo/sgf animation/step = \meta{number} or \meta{list of numbers} (initially 100)}
	Like |/pgfgo/sgf/step|.
\end{key}

\begin{key}{/pgfgo/sgf animation/restart = \meta{restart options} (default true)}
	Like |/pgfgo/sgf/restart|.
\end{key}

\begin{key}{/pgfgo/sgf animation/restart/true}
	Like |/pgfgo/sgf/restart/true|.
\end{key}

\begin{key}{/pgfgo/sgf animation/restart/false}
	Like |/pgfgo/sgf/restart/false|.
\end{key}

\begin{key}{/pgfgo/sgf animation/restart/auto}
	Like |/pgfgo/sgf/restart/auto|. It is the initial value.
\end{key}

\begin{key}{/pgfgo/sgf animation/restart/from = \meta{number} (initially 1)}
	Like |/pgfgo/sgf/restart/from|.
\end{key}

\begin{key}{/pgfgo/sgf animation/annotations = \meta{boolean} (default true, initially true)}
	With |comments|, it also writes the name and the annotations of each move, like |\sgfcomment|.
\end{key}

\begin{key}{/pgfgo/sgf animation/game = \meta{number} (initially 1)}
	Like |/pgfgo/sgf/game|.
\end{key}

\begin{key}{/pgfgo/sgf animation/view = \meta{boolean} (default true, initially true)}
	If the file has a view (|VW|), the boards only show that part, like |sgf view|. All the frames have the same board: if the view changes during the game, the board covers the views of all the frames (and if one of them shows the whole board, the whole board is shown).
\end{key}

\begin{key}{/pgfgo/sgf animation/layers = \meta{boolean} (default true, initially true)}
	Builds the animation by layers: the board is drawn only once, and so is every stone, on a transparency that |animate| shows while the stone stays on the board (with the |timeline| option of |animate|). The only things drawn in every frame are the things that change: the mark of the last move, the marks of the file and the comment. So the PDF grows with the number of moves and not with its square: a 200-move game takes less than 1~MB instead of more than 10~MB, and it also compiles faster. With |layers = false|, every frame is a whole |goban|. The |timeline| is saved in an auxiliary file, |<jobname>-pgfgo-<n>.tln|.
\end{key}

\begin{command}{\sgfdiagrams\opt{\oarg{options}}\marg{file}}
	Draws the game as books do: one diagram every |step| moves (1--50, 51--100, etc.). Each diagram starts from the real position, with the earlier moves without numbers and their captures; stones captured within the diagram are not removed (they are removed in the next one). Below each one go its caption and its "N at M" notes (p.~\pageref{subsec:notes}). It is used outside |goban|. The options can be set for the whole document with |\pgfgoset{sgf diagrams = {...}}|.

	\begin{codeexample}[code only]
		\sgfdiagrams[step = 5]{game.sgf}
	\end{codeexample}
\end{command}

\sgfdiagrams[step = 5]{game.sgf}

\begin{key}{/pgfgo/sgf diagrams/step = \meta{number} or \meta{list of numbers} (initially 50)}
	Number of moves of each diagram. With a list, the first diagram has as many moves as the first number, the second one as the second number, and so on, and the last number is repeated: with |step = {5, 10, 3}|, the diagrams have 5, 10, 3, 3, 3... moves.
\end{key}

\begin{key}{/pgfgo/sgf diagrams/from = \meta{number} (initially 1)}
	First move. The diagrams are the same as without |from|, with the same number: only the ones with moves from |from| on are drawn, and the first one starts at |from|. With |step = 50| and |from = 120|, the first one is "Diagram 3 (120--150)".
\end{key}

\begin{key}{/pgfgo/sgf diagrams/to = \meta{number} or |last| (initially last)}
	Last move. With |last| (or |-1|), the last one of the game.
\end{key}

\begin{key}{/pgfgo/sgf diagrams/numbers = \meta{boolean} (default true, initially true)}
	With |false|, the moves of each diagram are drawn without numbers (the "N at M" notes still use the numbers of the game).
\end{key}

\begin{key}{/pgfgo/sgf diagrams/last mark = \meta{mark options} or |false| (initially false)}
	Like |/pgfgo/sgf/last mark|, on each diagram.

	\begin{codeexample}[code only]
		\sgfdiagrams[step = 5, to = 10, numbers = false, last mark = circle]{game.sgf}

\subsection{Layout of the Diagrams}
\label{subsec:layout}

|\sgfdiagrams| and |\sgfproblem| draw several diagrams. By default they go in the paragraph, side by side, and move to the next line when they do not fit. With |columns|, they go in a grid of columns of the same width: each row is a separate line (the page can break between rows), the boards of a row are aligned at the top and an incomplete row leaves its last columns empty. With |fit|, the boards are also scaled to fill the width of their column.

\begin{codeexample}[code only]
	\sgfdiagrams[step = 2, to = 6, layout = {columns = 3, fit}]{game.sgf}
\end{codeexample}

\sgfdiagrams[step = 2, to = 6, layout = {columns = 3, fit}]{game.sgf}

\begin{stylekey}{/pgfgo/layout = \meta{layout options}}
	Layout options for the whole document, for example |\pgfgoset{layout = {columns = 3, fit}}|. For a single command, |/pgfgo/sgf diagrams/layout| and |/pgfgo/sgf problem/layout|; also for the whole document, |\pgfgoset{sgf diagrams = {layout = {...}}}|.
\end{stylekey}

\begin{key}{/pgfgo/layout/columns = \meta{number or auto} (initially auto)}
	Number of columns of the grid. With |auto|, the diagrams go in the paragraph, without a grid.
\end{key}

\begin{key}{/pgfgo/layout/width = \meta{dimension} (initially \texttt{\string\linewidth})}
	Width of the grid. If it is less than the width of the line, the grid is centered.
\end{key}

\begin{key}{/pgfgo/layout/column sep = \meta{dimension} (initially 1em)}
	Space between columns.
\end{key}

\begin{key}{/pgfgo/layout/row sep = \meta{skip} (initially \texttt{\string\medskipamount})}
	Space between rows.
\end{key}

\begin{key}{/pgfgo/layout/fit = \meta{boolean} (default true, initially false)}
	Scales each board to fill the width of its column; it replaces the |scale| of the board. The lines keep their width. In |\sgfdiagrams| and |\sgfproblem|, all the boards use the scale that fills the column with the first one, so the stones have the same size in all of them. Only with |columns|.
\end{key}

\begin{key}{/pgfgo/layout/per page = \meta{number} or |false| (initially 0)}
	Every \meta{number} diagrams, starts a new page (with |\newpage|): with |per page = 1|, one diagram per page. In the grid, a half-filled row is drawn before the break; \meta{number} should be a multiple of |columns|. With |0| or |false|, there are no breaks. In |\sgfproblem| with |statement = alone|, the statement counts as one diagram.

	\begin{codeexample}[code only]
		\sgfdiagrams[step = 50, layout = {columns = 2, fit, per page = 4}]{game.sgf}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/layout/per frame = \meta{number} or |false| (default 1, initially 0)}
	Inside a |beamer| |frame|, every \meta{number} diagrams go to the next slide: with |per frame| alone, one diagram per slide, each with its caption. The diagrams use |\only<+>|, so they add to the other overlays of the |frame|: after a |\pause|, the first diagram goes on the next slide. Outside |beamer|, the package warns and the key is not used. In |beamer|, if there is also |per page|, |per frame| wins.

	\begin{codeexample}[code only]
		\begin{frame}{The game}
			\sgfdiagrams[step = 50, goban = {scale = 0.6}, layout = {per frame}]{game.sgf}
		\end{frame}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/layout/align = \meta{center, left or right} (initially center)}
	Where each diagram goes within its column, with its caption and its notes. The comments are always aligned to the left.

	\begin{codeexample}[code only]
		\sgfdiagrams[step = 2, to = 6, layout = {columns = 2, width = 0.6\linewidth,
			column sep = 3em, row sep = 2em, align = left}]{game.sgf}
	\end{codeexample}

	\sgfdiagrams[step = 2, to = 6, layout = {columns = 2, width = 0.6\linewidth,
		column sep = 3em, row sep = 2em, align = left}]{game.sgf}
\end{key}

\begin{environment}{{pgfgolayout}\opt{\oarg{layout options}}}
	Lays out in a grid the |goban|s it contains, and the diagrams of the |\sgfdiagrams| and |\sgfproblem| it contains, in the order in which they appear. So boards drawn by hand and diagrams of different files can be put together in a grid. The options are those of |/pgfgo/layout|; the |layout| options of the commands inside are not used. There must be no text inside, only the boards and the commands. With |fit|, each |goban| is drawn several times to measure it: its contents should have no effects outside the board (for example, counters that are stepped).

	\begin{codeexample}[code only]
		\begin{pgfgolayout}[columns = 4, fit]
			\begin{goban}[5]\stones[b]{C3}\end{goban}
			\begin{goban}[5]\stones[w]{C3}\end{goban}
			\sgfdiagrams[step = 5]{game.sgf}
		\end{pgfgolayout}
	\end{codeexample}
\end{environment}

\begin{pgfgolayout}[columns = 4, fit]
	\begin{goban}[5]\stones[b]{C3}\end{goban}
	\begin{goban}[5]\stones[w]{C3}\end{goban}
	\sgfdiagrams[step = 5]{game.sgf}
\end{pgfgolayout}



\subsection{Prisoners}

\pgfgoname\ provides specific commands to represent prisoners, typically with reduced opacity to distinguish them from active stones.

\begin{command}{\prisoner\opt{\oarg{options}}\marg{intersection}}
	
	Draws a prisoner stone at a specific intersection, with options to customize its style.
\end{command}

\begin{command}{\prisoners\opt{\oarg{options}}\marg{intersection \opt{, intersection, ...}}}
	
	Places multiple prisoner stones in a single line, efficient for showing captured groups.
\end{command}

\begin{key}{/pgfgo/prisoner=\meta{stone options} (default fill opacity = 0.5)}
	
	Defines the characteristics of prisoner stones, with a default fill opacity of 0.5 to indicate their status. Other |\stone| options, such as color or size, can be applied. It applies to the prisoners of the three players; the style of each one (|/pgfgo/players/white/stone/prisoner|) comes after it.

	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = B2:D4}}

		\begin{goban}
			\stones[b]{C2}
			\prisoners[b]{D3}
			\prisoner[w]{B4}
		\end{goban}

		\pgfgoset{prisoner = {fill = {opacity = 0.2}, line = {color = red}}}

		\begin{goban}
			\stones[b]{C2}
			\prisoners[b]{D3}
			\prisoner[w]{B4}
		\end{goban}
	\end{codeexample}
\end{key}
