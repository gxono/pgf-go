\section{Marks}
\label{sec:marks}

Marks are graphical elements placed at the intersections of the board to highlight specific positions, indicate points of interest, or number sequences without the need to add stones. \pgfgoname\ offers versatile commands and options to customize these marks, ranging from predefined shapes to custom labels.

\begin{command}{\mark\opt{\oarg{options}}\marg{intersection}}
	
	Draws an individual mark at the specified intersection (e.g., "C3"). Options allow defining its shape, color, and other attributes.
\end{command}

\begin{command}{\marks\opt{\oarg{options}}\marg{intersection \opt{, intersection, ...}}}
	
	Similar to |\mark|, but applies the same mark to multiple intersections using the coordinate syntax.
\end{command}

\begin{key}{/pgfgo/mark = \meta{options} (default cross)}
	
	Defines the type of mark to be placed, with a cross ("cross") as the default value if no other option is specified.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = B2:D4}}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
		\end{goban}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
			\mark{C3}
		\end{goban}
	\end{codeexample}
\end{key}

	Allows numbering a series of marks in order.

			\stones[b]{C2,D3}
			\stone[w]{B4}

		\end{goban}

\begin{key}{/pgfgo/mark/sequence/true}
	Turns the sequence on. It is the same as |sequence| alone.
\end{key}

\begin{key}{/pgfgo/mark/sequence/false}
	Turns the sequence off (for example, one turned on with |\pgfgoset|).
\end{key}

\begin{key}{/pgfgo/mark/sequence/format = \meta{format}}
	Turns the sequence on and chooses the number format. It is the same as |sequence, label = {format = ...}| (see |label/format|).
\end{key}

\begin{key}{/pgfgo/mark/sequence/from=\meta{number}}
	
	Sets the starting number of a numbered mark sequence, allowing continuation from a specific point instead of 1.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = B2:D4}}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
		\end{goban}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
			\marks[sequence = {from = 3}]{C3,B3,C4}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/mark/dim}
	Dims the point: it covers it with the color of the background of the board, translucent, over the stone and the grid. It is what |\sgf| draws for the |DD| property.

	\begin{codeexample}[]
		\begin{goban}[5]
			\stones[b]{B2, C3}\stone[w]{D4}
			\marks[dim]{A1:5, B2}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/mark/custom=\meta{token list}}
	
	Allows defining a custom mark using text or symbols (e.g., "!" or "?") and even images.
	
	\begin{codeexample}[]
		\pgfgoset{goban = {5, partial = B2:D4}}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
		\end{goban}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
			\mark[custom = {!}]{C3}
		\end{goban}
		
		\begin{goban}
			\stones[b]{C2,D3}
			\stone[w]{B4}
			\mark[custom = {\tikz{\node[rounded corners, draw, inner sep = 1em, red]{\dots};}}]{C3}
		\end{goban}
	\end{codeexample}
\end{key}

\label{subsec:custommarks}\begin{command}{\pgfgodefinemark\opt{\oarg{options}}\marg{name}\marg{pgf code}}
	Defines a new mark, called \meta{name}, which is used like the others: in |\marks|, in the |mark| option of stones and players, and in |\territory|. It is drawn on the same layers as the other marks; on an empty point, it covers the grid with a circle of the color of the background. A mark with the name of another one replaces it.

	The \meta{pgf code} is drawn with the origin on the point. Coordinates without units (those of |\pgfpointxy| and of Ti\emph{k}Z) are measured in stone radii: |(0,0) circle (1)| has the size of the stone. The line color is |mark/line/color| of the player; the fill color, |mark/fill/color|; and the line width, |mark/line/width|. These two colors are also called |pgfgomarkline| and |pgfgomarkfill|. With |mark/scale|, the mark changes its size like the others.

	With |\pgfgodefinemark[image = {file = |\meta{file}|, width = |\meta{dimension}|}]|\marg{name}, without \meta{pgf code}, the mark is an image: see |image| below.

	\begin{codeexample}[]
		\pgfgodefinemark{diamond}{
			\pgfpathmoveto{\pgfpointxy{0}{0.7}}
			\pgfpathlineto{\pgfpointxy{0.7}{0}}
			\pgfpathlineto{\pgfpointxy{0}{-0.7}}
			\pgfpathlineto{\pgfpointxy{-0.7}{0}}
			\pgfpathclose
			\pgfusepath{fill, stroke}}
		\pgfgodefinemark{slash}{\draw (-0.5,-0.5) -- (0.5,0.5);}
		\pgfgodefinemark[image = {file = star.pdf}]{star}

		\begin{goban}[5, partial = B2:D4]
			\stones[b]{C2,D3}
			\stone[w]{B4}
			\marks[diamond]{C2, B4, B2}
			\marks[slash]{D3, C3}
			\marks[star]{D4}
		\end{goban}
	\end{codeexample}
\end{command}

\begin{key}{/pgfgo/define mark/image = \meta{options}}
	The image (PNG, JPG or PDF) of a mark defined with |\pgfgodefinemark|. It is drawn centered on the point; the colors of the marks do not apply.
\end{key}

\begin{key}{/pgfgo/define mark/image/file = \meta{file}}
	The file of the image.
\end{key}

\begin{key}{/pgfgo/define mark/image/width = \meta{dimension} (initially 1.2 stone radii)}
	The width of the image; the height keeps the proportions. Without this key, the mark has the size of the circle (|circle|), and changes with the radius of the stones.

	\begin{codeexample}[]
		\pgfgodefinemark[image = {file = star.pdf}]{star}
		\pgfgodefinemark[image = {file = star.pdf, width = 0.6em}]{starlet}

		\begin{goban}[5, partial = B2:D4]
			\stones[b]{C2,D3}
			\stone[w]{B4}
			\marks[star]{C3, B4}
			\marks[starlet]{B3, D3}
		\end{goban}
	\end{codeexample}
\end{key}


\begin{key}{/pgfgo/mark/scale = \meta{float} (initially 1)}
	The size of the mark compared with the usual one: the shapes, the images and the text.

	\begin{codeexample}[]
		\begin{goban}[5, partial = B2:D4]

			\marks[triangle, scale = 1.5]{C2, B3}
			\marks[circle, scale = 0.6]{D4}

	\end{codeexample}
\end{key}

\subsection{Arrows and Lines}
\label{subsec:arrows}

Arrows and lines join two intersections. They are drawn above the stones, with a halo in the color of the background of the board that separates them from the stones and the grid. Like |\mark|, |\arrow| and |\line| only exist inside |goban|: outside, |\line| is still the \LaTeX\ command. |\sgf| draws the arrows (|AR|) and lines (|LN|) of the file.

\begin{command}{\arrow\opt{\oarg{options}}\marg{from}\marg{to}}
	Draws an arrow from one intersection to another.

	\begin{codeexample}[]
		\begin{goban}[9]
			\stones[b]{C3, D3}\stone[w]{E5}
			\arrow{B3}{F3}
			\arrow[color = red, halo = false]{C7}{G9}
		\end{goban}
	\end{codeexample}
\end{command}

\begin{command}{\line\opt{\oarg{options}}\marg{from}\marg{to}}
	Draws a line from one intersection to another.

	\begin{codeexample}[]
		\begin{goban}[9]
			\stones[b]{C3, D4}\stone[w]{E5}
			\line{C3}{G7}
			\line[color = blue, width = 2pt]{A9}{J9}
		\end{goban}
	\end{codeexample}
\end{command}

\begin{key}{/pgfgo/arrow/color=\meta{color} (initially black)}
	Color of the arrow.
\end{key}

\begin{key}{/pgfgo/arrow/width=\meta{dimension} (initially 1.2pt)}
	Width of the arrow. The tip grows with it.
\end{key}

\begin{key}{/pgfgo/arrow/opacity=\meta{number} (initially 1)}
	Opacity of the arrow.

	\begin{codeexample}[]
		\begin{goban}[5]
			\stones[b]{C2, C4}\stone[w]{D3}
			\arrow[width = 0.4pt]{A1}{E1}
			\arrow[color = blue, width = 2.5pt]{A3}{E3}
			\arrow[color = red, opacity = 0.4, width = 2.5pt]{A5}{E5}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/arrow/dash=\meta{solid, dashed or dotted} (initially solid)}
	The stroke: solid, dashed or dotted. The halo and the tip are always solid.
\end{key}

\begin{key}{/pgfgo/arrow/shorten=\meta{dimension} (initially 0pt)}
	Shortens both ends, for example so that the arrow does not cover the stones.

	\begin{codeexample}[]
		\begin{goban}[5]
			\stones[b]{B2}\stones[w]{D4}
			\arrow[shorten = 5pt]{B2}{D4}
			\arrow[dash = dashed, tips = both]{A5}{E5}
			\line[dash = dotted]{A1}{E1}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/arrow/bend=\meta{angle} (initially 0)}
	Draws an arc instead of a straight line: the arrow leaves turned that angle to the left and arrives turned the same, like |bend left| in \tikzname. With a negative angle, it bends to the right.

	\begin{codeexample}[]
		\begin{goban}[5]
			\stones[b]{B2}\stones[w]{D4}
			\arrow[bend = 40, shorten = 5pt]{B2}{D4}
			\line[bend = -30]{A1}{E1}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/arrow/tip=\meta{arrow tip} (initially Stealth)}
	The tip of the arrow, one of the tips of the |arrows.meta| library of \tikzname, like |Latex| or |Triangle|.
\end{key}

\begin{key}{/pgfgo/arrow/tips=\meta{end or both} (initially end)}
	Where the tip goes: only at the end, or at both ends.

	\begin{codeexample}[]
		\begin{goban}[5]
			\arrow[tip = Latex]{A1}{E1}
			\arrow[tip = Triangle, tips = both]{A3}{E3}
			\arrow[tip = {Circle[open]}]{A5}{E5}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/arrow/halo=\meta{halo options} (default true)}
	The halo around the arrow.
\end{key}

\begin{stylekey}{/pgfgo/arrow/halo/true}
	Draws the halo. It is the initial value.
\end{stylekey}

\begin{stylekey}{/pgfgo/arrow/halo/false}
	Does not draw the halo.
\end{stylekey}

\begin{key}{/pgfgo/arrow/halo/color=\meta{color}}
	Color of the halo. Without this \emph{key}, it is the color of the background of the board.
\end{key}

\begin{key}{/pgfgo/arrow/halo/width=\meta{dimension} (initially 0.8pt)}
	Width of the halo on each side of the arrow.

	\begin{codeexample}[]
		\begin{goban}[5]
			\stones[b]{B1, B3, B5}\stones[w]{D1, D3, D5}
			\arrow[halo = false]{A1}{E1}
			\arrow{A3}{E3}
			\arrow[halo = {color = yellow, width = 2pt}]{A5}{E5}
		\end{goban}
	\end{codeexample}
\end{key}

\begin{key}{/pgfgo/arrow/forget}
	|resume| does not bring this arrow, even with |marks|.
\end{key}
