\section{First steps}

\begin{frame}{What pgf-go is}
	\begin{columns}[c]
		\column{0.6\textwidth}
			\small
			\begin{itemize}[<+->]
				\setlength\itemsep{0.5pt}
				\item Go boards and positions, in color, for documents and slides.
				\item Games in SGF: one board, diagrams as in books, animations and problems.
				\item The rules, with the Lua kernel: captures, ``N at M'' notes, territory and score.
				\item Visual profiles, with images: wood, slate and shell.
				\item Boards with a name: variations and views in the middle of the text.
				\item Marks, arrows, lines and custom marks.
				\item A grid to lay out diagrams.
				\item Texts in ten languages and Western, Japanese or Chinese coordinates.
				\item A syntax to write many coordinates with few characters.
			\end{itemize}
		\column{0.35\textwidth}
			\centering
			\begin{goban}[9, profile = yunzi, scale = 1.05]
				\stones[b]{C3, D4, E5, G3, C7, F7}
				\stones[w]{G7, F4, D6, C5, E3}
				\marks[b, triangle]{E5}
				\arrow[color = crimson]{G3}{G5}
			\end{goban}
	\end{columns}
\end{frame}

\begin{frame}[fragile]{A minimal document}
	\begin{columns}[c]
		\column{0.5\textwidth}

\begin{frame}[fragile]{Two kernels: expl3 and Lua}
	The computation (coordinates, turns, numbering, memory) is done by a \emph{kernel}; the drawing is always the same.

	\medskip
	\begin{columns}[t]
		\column{0.55\textwidth}
			\renewcommand\arraystretch{1.25}
			\begin{tabular}{lcc}
				& \keyname{expl3} & \keyname{lua} \\\hline
				Engine & any & Lua\LaTeX \\
				Stones, moves, marks & \checkmark & \checkmark \\
				Memory and views & \checkmark & \checkmark \\
				Captures (\keyname{captures}) & & \checkmark \\
				\keyname{\textbackslash territory\{auto\}} & & \checkmark \\
				Games in SGF & & \checkmark \\
			\end{tabular}
		\column{0.4\textwidth}
			With Lua\LaTeX, the Lua kernel is used on its own. To choose the other one:
			\begin{codeonly}
\usepackage[kernel = expl3]{pgf-go}
			\end{codeonly}
			Both give the same diagrams.
	\end{columns}
\end{frame}

\begin{frame}[fragile]{\texttt{\textbackslash pgfgoset} and the keys that turn on and off}
	\begin{codebox}[0.6]
\pgfgoset{goban = {5, partial = B2:D4}}

% The key alone (or true) turns on.
\begin{goban}[stone = {shine}]
	\stones[b]{C2, D3} \stone[w]{B4}
\end{goban}
% Giving options does not turn on...
\begin{goban}[stone = {shine = {angle = 45}}]
	\stones[b]{C2, D3} \stone[w]{B4}
\end{goban}
% ...except with true.
\begin{goban}[stone = {shine = {true, angle = 45}}]
	\stones[b]{C2, D3} \stone[w]{B4}
\end{goban}
	\end{codebox}
	\small What \keyname{\textbackslash pgfgoset} sets applies to everything that follows; each \keyname{goban} can change it.
\end{frame}

\begin{frame}{How pgf-go stores images in the PDF}
	\begin{columns}[c]
		\column{0.5\textwidth}
			\begin{itemize}[<+->]
				\item With pdf\LaTeX\ or Lua\LaTeX, each image is stored \alert{only once}, the first time it is
					used, in a \emph{Form XObject}.
				\item After that, each stone, tile or mark that uses it only \emph{names} it: a few bytes.
				\item Rotating it (\keyname{rotate}) does not add another copy, and the shadow with its shape uses the same copy.
				\item What does add a copy is another size: another \keyname{width}, or another \keyname{scale}.
				\item With Xe\LaTeX\ or DVI output, the image is included every time.
			\end{itemize}
		\column{0.45\textwidth}
			\centering
			\begin{tikzpicture}[x = 1em, y = 1em, font = \scriptsize]
				\node[draw = walnut, fill = sand, rounded corners, align = center, minimum width = 9em, inner sep = 0.5em]
					(form) at (0, 0) {\includegraphics[width = 2.2em]{pgfgo-shell-white-2.png}\\[2pt]
						\texttt{pgfgo-shell-white-2.png}\\one copy in the PDF};
				\foreach \x/\y [count = \k] in {-6/-6, -2/-6.6, 2/-6.6, 6/-6} {
					\node (p\k) at (\x, \y) {\includegraphics[width = 1.6em]{pgfgo-shell-white-2.png}};
					\draw[->, walnut!70] (p\k) -- (form);
				}
				\node[text = walnut] at (0, -8.6) {hundreds of stones: one reference each};
			\end{tikzpicture}
	\end{columns}
\end{frame}

\begin{frame}{What makes a PDF bigger and what does not}
	\begin{columns}[t]
		\column{0.48\textwidth}
			{\color{walnut}\bfseries It does not make it bigger}
			\begin{itemize}
				\item More stones with the same image and the same size.
				\item Rotating the images, or choosing among several files.
				\item The shadow with the shape of the image.
				\item Many marks on a background of tiles: only the tiles the mark touches are redrawn.
				\item An animation in layers (\keyname{layers}).
			\end{itemize}
		\column{0.48\textwidth}
			{\color{crimson}\bfseries It makes it bigger}
			\begin{itemize}
				\item The same image with another size: another \keyname{width}, another \keyname{scale}, \keyname{fit} in
					different grids.
				\item Translucent stones (\keyname{opacity} less than 1): each one goes in a transparency group.
				\item Animations with \keyname{layers = false}: a whole board per frame.
				\item Xe\LaTeX\ or DVI output with images.
			\end{itemize}
	\end{columns}
	\vfill
	\small In \keyname{\textbackslash sgfdiagrams} and \keyname{\textbackslash sgfproblem} with \keyname{fit}, all the boards use the same scale: the stones have the same size, and each image is stored once.
\end{frame}
