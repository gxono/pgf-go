\documentclass{article}

\newcommand\case[1]{\par\medskip\noindent\textbf{#1}\par\nopagebreak}
\pagestyle{empty}
\begin{document}

\case{name / resume / a plain goban stays empty}
\begin{goban}[5, name = a]\stones[b]{C3}\stone[w]{D4}\end{goban}
\begin{goban}[5, resume = a]\end{goban}
\begin{goban}[5]\end{goban}

\case{stone[forget] is not resumed and does not affect the next stones (resume: B2, D2)}
\begin{goban}[5, name = a]\stone[b, forget]{C3}\stone[b]{B2}\stones[w]{D2}\end{goban}
\begin{goban}[5, resume = a]\end{goban}

\case{Variations: b = a + E5, c = a + A1; a does not change (resume a: only C3)}
\begin{goban}[5, name = a]\stone[b]{C3}\end{goban}
\begin{goban}[5, name = b, resume = a]\stone[w]{E5}\end{goban}
\begin{goban}[5, resume = a, name = c]\stone[w]{A1}\end{goban}
\begin{goban}[5, resume = a]\end{goban}
\begin{goban}[5, resume = b]\end{goban}

\case{name = a, resume = a keeps adding to a (C3, E5, A5)}
\begin{goban}[5, name = a]\stone[b]{C3}\end{goban}
\begin{goban}[5, name = a, resume = a]\stone[w]{E5}\end{goban}
\begin{goban}[5, resume = a, name = a]\stone[w]{A5}\end{goban}
\begin{goban}[5, resume = a]\end{goban}

\case{Moves: no numbers by default; numbering and turn continue (4 white, 5 black); numbers; label = forget hides 2}
\begin{goban}[5, name = a]\moves*{C2}\moves*[label = forget]{B4}\moves*{D3}\end{goban}
\begin{goban}[5, resume = a]\moves*{C:D4}\end{goban}
\begin{goban}[5, resume = {a, numbers}]\moves*{C:D4}\end{goban}

\case{moves[forget] is not resumed (resume: only C3, and the next move is 1)}
\begin{goban}[5, name = a]\stone[b]{C3}\moves*[forget]{D3,D4}\end{goban}
\begin{goban}[5, resume = a]\moves*{B2}\end{goban}

\case{Marks only with marks; sequences restart at 1; marks[forget] is not resumed (no square on B2)}
\begin{goban}[5, name = a]\stones[b]{C3}\marks[b, triangle]{C3}\marks[sequence]{A1,B1,C1}\marks[forget, square]{B2}\end{goban}
\begin{goban}[5, resume = a]\end{goban}
\begin{goban}[5, resume = {a, marks}]\end{goban}

\case{Territory only with territory}
\begin{goban}[5, name = a]\stones[b]{B1,B2,A3,B3}\territory[b]{A1,A2}\end{goban}
\begin{goban}[5, resume = a]\end{goban}
\begin{goban}[5, resume = {a, territory}]\end{goban}

\case{b keeps what it got from a as it was drawn (no numbers, no marks) plus its own mark and move}
\begin{goban}[5, name = a]\moves*{C3,D3}\marks[circle]{A1}\end{goban}
\begin{goban}[5, name = b, resume = a]\moves*{C4}\marks[triangle]{E5}\end{goban}
\begin{goban}[5, resume = {b, numbers, marks}]\end{goban}

\case{Fragile options survive the memory}
\begin{goban}[5, name = a]\stone[w, mark = {custom = \textbf{A}}]{C3}\marks[custom = \emph{x}]{B2}\end{goban}
\begin{goban}[5, resume = {a, marks}]\end{goban}

\case{resume on a partial view of a full board}
\begin{goban}[9, name = a]\stones[b]{C3,G7}\stones[w]{D3,G6}\end{goban}
\begin{goban}[9, resume = a, partial = A1:E5]\end{goban}

\case{Views: what was forgotten is not drawn, unless forgotten}
\begin{goban}[5, name = f]\stones[b]{C3}\stone[w, forget]{D4}\moves*[label = forget]{B2}\marks[forget, circle]{A1}\end{goban}
\pgfgoview{f}{A1:E5} \pgfgoview[forgotten]{f}{A1:E5}

\case{resume keeps the size of the named board (13, then 7x5 through two names); a size of its own wins (15)}
\begin{goban}[13, scale = 0.4, name = big]\stones[b]{K10, C3}\end{goban}
\begin{goban}[scale = 0.4, resume = big]\end{goban}
\begin{goban}[scale = 0.4, resume = big, 15]\end{goban}
\begin{goban}[7x5, scale = 0.4, name = rect]\stones[w]{G5}\end{goban}
\begin{goban}[scale = 0.4, resume = rect, name = rect2]\stones[b]{A1}\end{goban}
\begin{goban}[scale = 0.4, resume = rect2]\end{goban}
