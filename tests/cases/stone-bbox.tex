	\begin{goban}[5, label = {at = {south, west}}]\stones[b]{E5}\territory[dame]{C3}\end{goban}}
\newsavebox\before
\newsavebox\other
\newsavebox\after
\sbox\before{\gobanE}
\sbox\other{\begin{goban}[5, label = {at = {south, west}}]\stones[b]{A1}\end{goban}}
\sbox\after{\gobanE}

\case{The same goban, before and after another one}
