% The frames cannot be rendered inside the animation, so \frames draws them
% one by one with the same functions: \frames{<options>}{<file>}{<frames>}.
\ExplSyntaxOn
\NewDocumentCommand \frames {mmm} {
	\group_begin:
		\pgfgo_sgfanimation_setup:nn {#1} {#2}
		\clist_map_inline:nn {#3} {\pgfgo_sgfanimation_frame:n {##1}\quad}
	\group_end:
}

% Every frame of an animation must have the same size, or animate stretches
% some of them: \samesize{<options>}{<file>} checks all the frames.
\msg_new:nnn {sgf-animation-test} {size}
	{Frame~#1~is~#2,~but~frame~0~is~#3.}
\tl_new:N \l__test_size_tl
\tl_new:N \l__test_first_size_tl
\NewDocumentCommand \samesize {mm} {
	\group_begin:
		\pgfgo_sgfanimation_setup:nn {#1} {#2}
		\int_step_inline:nnn {0} {\seq_count:N \l__pgfgo_sgfanim_moves_seq} {
			\hbox_set:Nn \l_tmpa_box {\pgfgo_sgfanimation_frame:n {##1}}
			\tl_set:Ne \l__test_size_tl {
				\dim_use:N \box_wd:N \l_tmpa_box ~ x ~
				\dim_eval:n {\box_ht:N \l_tmpa_box + \box_dp:N \l_tmpa_box}
			}
			\int_compare:nNnTF {##1} = {0}
				{\tl_set_eq:NN \l__test_first_size_tl \l__test_size_tl}
				{
					\tl_if_eq:NNF \l__test_size_tl \l__test_first_size_tl
						{\msg_error:nneee {sgf-animation-test} {size} {##1} {\l__test_size_tl} {\l__test_first_size_tl}}
				}
		}
	\group_end:
}
\ExplSyntaxOff
