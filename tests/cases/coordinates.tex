\documentclass{article}
\usepackage[margin=1cm]{geometry}
\usepackage{../../latex/pgf-go}
\newcommand\case[1]{\par\medskip\noindent\textbf{#1}\par\nopagebreak}
\pagestyle{empty}
\begin{document}

\case{Whole column (B), whole row (4), star on a partial board}
\begin{goban}[5]\stones[b]{B}\end{goban}
\begin{goban}[5]\stones[w]{4}\end{goban}
\begin{goban}[7, partial = {from = B2 to D4}]\stones[b]{*}\end{goban}

\case{Ranges: A:B (columns), 2:3 (rows), B:D 4, C 1:3, A:B 1:2}
\begin{goban}[5]\stones[b]{A:B}\end{goban}
\begin{goban}[5]\stones[b]{2:3}\end{goban}
\begin{goban}[5]\stones[b]{B:D 4}\end{goban}
\begin{goban}[5]\stones[b]{C 1:3}\end{goban}
\begin{goban}[5]\stones[b]{A:B 1:2}\end{goban}

\case{A:B on a partial board that is not square (A1 to E3): only rows 1-3}
\begin{goban}[9, partial = {from = A1 to E3}]\stones[b]{A:B}\end{goban}

\case{Operators: A1 -| C3, A1 |- C3, A2 -- E2, B1 r D3, B1 R D3}
\begin{goban}[5]\stones[b]{A1 -| C3}\end{goban}
\begin{goban}[5]\stones[b]{A1 |- C3}\end{goban}
\begin{goban}[5]\stones[b]{A2 -- E2}\end{goban}
\begin{goban}[5]\stones[b]{B1 r D3}\end{goban}
\begin{goban}[5]\stones[b]{B1 R D3}\end{goban}

\case{Chained operators: A1 R B2 r A4 |- C5 -- E5}
\begin{goban}[5]\stones[b]{A1 R B2 r A4 |- C5 -- E5}\end{goban}

\case{Brackets: A[1,3:4], [B,D]2, [A:B][4:5]}
\begin{goban}[5]\stones[b]{A[1,3:4]}\end{goban}
\begin{goban}[5]\stones[b]{[B,D]2}\end{goban}
\begin{goban}[5]\stones[b]{[A:B][4:5]}\end{goban}

\case{Additive and subtractive groups: rectangle A1 R E5 minus C3 and row 1}
\begin{goban}[5]\stones[b]{A1 R E5; C3, 1}\end{goban}

\case{Rows and stars do not reuse the coordinates of the previous command}
\begin{goban}[5]\stone[b]{C3}\marks[circle]{1}\end{goban}

\end{document}
