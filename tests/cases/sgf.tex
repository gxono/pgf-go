% game.sgf: 9x9, black C7, C6, C5 (C6 and C5 from a compressed list), white
% G3; then B E5, W G7, B pass, W C3, B G5. A variation (B A9) is not drawn.

\case{The whole game (move 3 is a pass, so there is no 3)}
\begin{goban}[9]\sgf{game.sgf}\end{goban}

\case{from = 4: moves 1 and 2 without numbers / to = 2 / the game continues with moves (6 white)}
\begin{goban}[9]\sgf[from = 4]{game.sgf}\end{goban}
\begin{goban}[9]\sgf[to = 2]{game.sgf}\end{goban}
\begin{goban}[9]\sgf{game.sgf}\moves*{E7}\end{goban}

\case{Partial board, with captures and marks; resume}
\begin{goban}[9, captures, new remember, partial = {from = A1 to E9}, label = {at = all}]
	\sgf{game.sgf}\marks[b, triangle]{C6}
\end{goban}
\begin{goban}[9, resume]\end{goban}

\case{setup.sgf: AB, AW and AE after the first move. With captures, C3 is removed before black 4 plays there / up to move 3 / without captures AE cannot be applied: one warning}
\begin{goban}[5, captures]\sgf{setup.sgf}\end{goban}
\begin{goban}[5, captures]\sgf[to = 3]{setup.sgf}\end{goban}
\begin{goban}[5]\sgf{setup.sgf}\end{goban}

\case{variations.sgf (7x7, sgf size): up to move 2 with its marks (square and circle on stones, cross and labels A and B on empty points) / marks = false / variation = 2, with a triangle on 3 / variation = 1.2}
\begin{goban}[sgf size = variations.sgf]\sgf[to = 2]{variations.sgf}\end{goban}
\begin{goban}[sgf size = variations.sgf]\sgf[to = 2, marks = false]{variations.sgf}\end{goban}
\begin{goban}[sgf size = variations.sgf]\sgf[variation = 2]{variations.sgf}\end{goban}
\begin{goban}[sgf size = variations.sgf]\sgf[variation = 1.2]{variations.sgf}\end{goban}

\case{MN[10] on move 4: 1, 2, 3, 10, 11 / restart = 3: 1, 2, 3, 1, 2 / variation = 3 does not exist: one warning, main line}
\begin{goban}[sgf size = variations.sgf]\sgf{variations.sgf}\end{goban}
\begin{goban}[sgf size = variations.sgf]\sgf[restart = 3]{variations.sgf}\end{goban}
\begin{goban}[sgf size = variations.sgf]\sgf[variation = 3]{variations.sgf}\end{goban}

\case{sgfinfo: PB = \sgfinfo{variations.sgf}{PB}, PW = \sgfinfo{variations.sgf}{PW}, RE = \sgfinfo{variations.sgf}{RE}, KM = \sgfinfo{variations.sgf}{KM}, DT = \sgfinfo{variations.sgf}{DT}, missing XX = [\sgfinfo{variations.sgf}{XX}]}
