\documentclass{article}
\usepackage[margin=1cm]{geometry}
\usepackage{../../latex/pgf-go}
\newcommand\case[1]{\par\medskip\noindent\textbf{#1}\par\nopagebreak}
\pagestyle{empty}
\begin{document}

\case{Sizes: 3, 9, 5x7 and a goban option does not leak into the next goban}
\begin{goban}[3]\end{goban}
\begin{goban}[5x7]\end{goban}
\begin{goban}\end{goban}

\case{Labels: all sides, font size 6pt, bold font, text colour}
\begin{goban}[5, label = {at = all}]\end{goban}
\begin{goban}[5, label = {at = {south, west}, font size = 6pt}]\end{goban}
\begin{goban}[5, label = {at = {south, west}, font = \bfseries, color = red}]\end{goban}

\case{Labels without I (the default) and with use i}
\begin{goban}[9, label = {at = {south}}]\end{goban}
\begin{goban}[9, use i, label = {at = {south}}]\end{goban}

\case{Partial boards hide what is outside the window (only C3 visible)}
\begin{goban}[5, partial = B2:D4]\stones[b]{A1,E5}\stones[w]{C3}\marks[circle]{A5}\moves{A2,C2}\end{goban}

\case{A partial window larger than the board is clipped}
\begin{goban}[3, partial = B2:D4]\stones[b]{*}\end{goban}

\case{Grid and background: false, then true again after a global false}
\begin{goban}[5, grid = false]\stone{C3}\end{goban}
\begin{goban}[5, background = false]\stone{C3}\end{goban}
\pgfgoset{goban = {background = false}}
\begin{goban}[5, background = true]\stone{C3}\end{goban}
\pgfgoset{goban = {background = true}}

\case{Background options: rounded corners, shadow, fill colour}
\begin{goban}[5, background = {rounded corners = {outer = 6pt}, shadow = true, fill = {color = orange!30}}]\end{goban}

\case{Scale 0.6}
\begin{goban}[5, scale = 0.6]\stones[b]{C3}\end{goban}

\end{document}
