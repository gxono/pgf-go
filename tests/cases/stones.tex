\documentclass{article}
\usepackage[margin=1cm]{geometry}
\usepackage{../../latex/pgf-go}
\newcommand\case[1]{\par\medskip\noindent\textbf{#1}\par\nopagebreak}
\pagestyle{empty}
\begin{document}

\case{Default players: \textbackslash stone is neutral (C3), \textbackslash stones is black (B2, D2)}
\begin{goban}[5]\stone{C3}\stones[b]{B2,D2}\end{goban}

\case{Explicit players: b, w, n and their long names}
\begin{goban}[5]
	\stones[b]{A1}\stones[w]{B1}\stones[n]{C1}
	\stone[black]{A2}\stone[white]{B2}\stone[neutral]{C2}
\end{goban}

\case{Stone options: radius 3pt (B2), scale 1.4 (C3), red 2pt line (D4)}
\begin{goban}[5]
	\stones[b, radius = 3pt]{B2}
	\stones[b, scale = 1.4]{C3}
	\stones[w, line = {color = red, width = 2pt}]{D4}
\end{goban}

\case{Shine and shadow with the default profile}
\begin{goban}[5]
	\stones[b, shine = {true}]{B3}
	\stones[w, shadow = {true, offset = 1pt, angle = -45}]{D3}
\end{goban}

\case{Prisoners: global red option (C3), per-stone blue option (D3), \textbackslash prisoner (B3)}
\pgfgoset{prisoner = {fill = {color = red}}}
\begin{goban}[5]
	\prisoners[w]{C3}
	\stones[b, prisoner = {fill = {color = blue}}]{D3}
	\prisoner[b]{B3}
\end{goban}
\pgfgoset{prisoner = {}}

\case{Prisoner style of white only (red line): white stone (B2) and black prisoner (C2) unchanged, white prisoner (D2) red}
\pgfgoset{players = {white = {stone = {prisoner = {line = {color = red}}}}}}
\begin{goban}[5]
	\stones[w]{B2}
	\prisoner[b]{C2}
	\prisoner[w]{D2}
\end{goban}
\pgfgoset{players = {white = {stone = {prisoner = {}}}}}

\case{Without use i (the default), J is the 9th column}
\begin{goban}[9, label = {at = south}]\stone[b]{J1}\stones[w]{J2}\end{goban}

\case{Stone with a mark: white stone with triangle, black stone with circle}
\begin{goban}[5]\stone[w, mark = triangle]{B2}\stone[b, mark = circle]{D2}\end{goban}

\end{document}
